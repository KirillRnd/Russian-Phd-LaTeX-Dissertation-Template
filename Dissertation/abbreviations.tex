\chapter*{Основные обозначения и сокращения}
\addcontentsline{toc}{chapter}{Основные обозначения и сокращения}

\begin{description}
	\item[CSV] текстовый формат представления табличных данных;
	\item[JSON] текстовый формат обмена структурированными данными;
	\item[OCR] оптическое распознавание текста;
	\item[TF--IDF] статистическая мера значимости термина в документе относительно корпуса;
	\item[URL] унифицированный указатель ресурса.
\end{description}
