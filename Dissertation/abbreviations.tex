\chapter*{Основные обозначения и сокращения}
\addcontentsline{toc}{chapter}{Основные обозначения и сокращения}

\section*{Сокращения}

\begin{description}
	\item[ЭРДУ] электрореактивная двигательная установка.
	\item[КА] космический аппарат.
	\item[ИР] идеально регулируемый.
	\item[ПСИ] постоянная скорость истечения.
	\item[КЛО] кривая локальных оптимумов.
\end{description}

\section*{Основные обозначения}

\begin{description}
	\item[$t$, $t_0$, $t_f$] время, начальный и конечный моменты времени.
	\item[$\mathbf r$, $r$] радиус-вектор положения КА и его длина.
	\item[$\mathbf v$] вектор скорости КА.
	\item[$\mathbf a$] реактивное ускорение.
	\item[$\mathbf e$] единичный вектор направления тяги.
	\item[$\mu$] гравитационный параметр центрального тела.
	\item[$\nu$] постоянная скорость истечения рабочего тела.
	\item[$\mathbf p_r$] сопряжённый положению вектор.
	\item[$\mathbf p_v$] сопряжённый скорости вектор.
	\item[$\mathbf z_0$] начальный вектор сопряжённых переменных: $\mathbf z_0=(\mathbf p_{r0},\mathbf p_{v0})$.
	\item[$T$] продолжительность перелёта.
	\item[$J$] значение квадратичного функционала.
	\item[$\boldsymbol\xi$] вектор внешних параметров задачи.
	\item[$\varphi$] угловая дальность.
	\item[$\Phi$] синодическая фаза в радианах.
	\item[$\Psi$] нормированная непрерывная фаза.
	\item[$R$] матрица линейного преобразования или $R$-преобразование.
	\item[$\mathbf q$] вектор управляющих параметров: $\mathbf q=(\lambda,\kappa,\zeta,\chi,j,\gamma)$.
	\item[$\lambda$] витковая дальность.
	\item[$\kappa$] отношение радиусов модельных круговых орбит.
	\item[$\zeta$] управляющий параметр $R$-преобразования II типа.
	\item[$\chi$] управляющий угол плоского $R$-преобразования для задания долготы перицентра.
	\item[$j$] управляющий параметр $R$-преобразования III типа.
	\item[$\gamma$] управляющий угол пространственного $R$-преобразования для задания долготы восходящего узла.
	\item[$e$] эксцентриситет конечной орбиты.
	\item[$a$] большая полуось конечной орбиты.
	\item[$i$] наклонение конечной орбиты.
	\item[$\Omega$] долгота восходящего узла.
	\item[$\bar\omega$] долгота перицентра.
	\item[$\omega$] аргумент перицентра.
	\item[$\widehat{x}$] аппроксимация величины $x$.
	\item[$x^*$] значение величины $x$, соответствующее численно найденному оптимальному решению.
	\item[$\mathcal A^{(x)}$] коэффициент-функция конечной формулы для величины $x$.
	\item[$C^{(x)}$] числовой коэффициент конечной формулы для величины  $x$.
\end{description}
