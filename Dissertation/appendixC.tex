\section*{Коэффициент-функция чувствительности для \(R\)-преобразования III типа}

Коэффициент-функция чувствительности, связывающая управляющий параметр \(j\) с наклонением конечной орбиты, имеет вид
\begin{equation}
\mathcal A^{(i)}(\kappa,\Omega)
=
\mathcal A_0^{(i)}(\kappa)+\mathcal A_1^{(i)}(\kappa)\cos 2\Omega,
\label{eq:appendixC_Ai}
\end{equation}
где
\begin{equation}
\mathcal A_0^{(i)}(\kappa)
=
\kappa
\left(C_0^{(i)}+C_1^{(i)}\kappa\right)
\left(\kappa+C_2^{(i)}\right)
+C_3^{(i)},
\label{eq:appendixC_Ai_0}
\end{equation}
\begin{equation}
\mathcal A_1^{(i)}(\kappa)
=
C_4^{(i)}
\left(\kappa-C_5^{(i)}\right)
\left(\kappa^3-C_6^{(i)}\right).
\label{eq:appendixC_Ai_1}
\end{equation}

\section*{Вспомогательные функции для коэффициентов параметров оптимального управления}

\noindent
Вспомогательные функции \(\mathcal P_i^{(\kappa)}(\lambda,\kappa_1,\kappa_2)\) рассчитываются на основе эталонных значений сопряжённых переменных для двух соотношений радиусов. Они не являются числовыми коэффициентами вида \(C_i^{(X)}\), поэтому записываются каллиграфической буквой.

\begin{equation}
\mathcal P_0^{(\kappa)}(\lambda,\kappa_1,\kappa_2)=
\frac{\kappa_1^{3/4}\kappa_2^{3/4}}{\kappa_2^{3/4}-\kappa_1^{3/4}}
\left(
\kappa_1^{1/4}\frac{p_{rx}(\lambda,\kappa_1)}{\ln\kappa_1}-
\kappa_2^{1/4}\frac{p_{rx}(\lambda,\kappa_2)}{\ln\kappa_2}
\right).
\label{eq:appendixC_P0_kappa}
\end{equation}

\begin{equation}
\mathcal P_1^{(\kappa)}(\lambda,\kappa_1,\kappa_2)=
\frac{1}{\kappa_1^{3/4}-\kappa_2^{3/4}}
\left(
\kappa_1\frac{p_{rx}(\lambda,\kappa_1)}{\ln\kappa_1}-
\kappa_2\frac{p_{rx}(\lambda,\kappa_2)}{\ln\kappa_2}
\right).
\label{eq:appendixC_P1_kappa}
\end{equation}

\begin{equation}
\mathcal P_2^{(\kappa)}(\lambda,\kappa_1,\kappa_2)=
\frac{\kappa_1^{1/4}\kappa_2^{1/4}}{\kappa_2^{1/4}-\kappa_1^{1/4}}
\left(
\kappa_1^{3/4}\frac{p_{ry}(\lambda,\kappa_1)}{\ln\kappa_1}-
\kappa_2^{3/4}\frac{p_{ry}(\lambda,\kappa_2)}{\ln\kappa_2}
\right).
\label{eq:appendixC_P2_kappa}
\end{equation}

\begin{equation}
\mathcal P_3^{(\kappa)}(\lambda,\kappa_1,\kappa_2)=
\frac{1}{\kappa_1^{1/4}-\kappa_2^{1/4}}
\left(
\kappa_1\frac{p_{ry}(\lambda,\kappa_1)}{\ln\kappa_1}-
\kappa_2\frac{p_{ry}(\lambda,\kappa_2)}{\ln\kappa_2}
\right).
\label{eq:appendixC_P3_kappa}
\end{equation}

\begin{equation}
\mathcal P_4^{(\kappa)}(\lambda,\kappa_1,\kappa_2)=
\frac{\kappa_1^{1/4}\kappa_2^{1/4}}{\kappa_2^{1/4}-\kappa_1^{1/4}}
\left(
\kappa_1^{3/4}\frac{p_{vx}(\lambda,\kappa_1)}{\ln\kappa_1}-
\kappa_2^{3/4}\frac{p_{vx}(\lambda,\kappa_2)}{\ln\kappa_2}
\right).
\label{eq:appendixC_P4_kappa}
\end{equation}

\begin{equation}
\mathcal P_5^{(\kappa)}(\lambda,\kappa_1,\kappa_2)=
\frac{1}{\kappa_1^{1/4}-\kappa_2^{1/4}}
\left(
\kappa_1\frac{p_{vx}(\lambda,\kappa_1)}{\ln\kappa_1}-
\kappa_2\frac{p_{vx}(\lambda,\kappa_2)}{\ln\kappa_2}
\right).
\label{eq:appendixC_P5_kappa}
\end{equation}

\begin{equation}
\mathcal P_6^{(\kappa)}(\lambda,\kappa_1,\kappa_2)=
\frac{\kappa_1^{3/4}\kappa_2^{3/4}}{\kappa_2^{3/4}-\kappa_1^{3/4}}
\left(
\kappa_1^{1/4}\frac{p_{vy}(\lambda,\kappa_1)}{\ln\kappa_1}-
\kappa_2^{1/4}\frac{p_{vy}(\lambda,\kappa_2)}{\ln\kappa_2}
\right).
\label{eq:appendixC_P6_kappa}
\end{equation}

\begin{equation}
\mathcal P_7^{(\kappa)}(\lambda,\kappa_1,\kappa_2)=
\frac{1}{\kappa_1^{3/4}-\kappa_2^{3/4}}
\left(
\kappa_1\frac{p_{vy}(\lambda,\kappa_1)}{\ln\kappa_1}-
\kappa_2\frac{p_{vy}(\lambda,\kappa_2)}{\ln\kappa_2}
\right).
\label{eq:appendixC_P7_kappa}
\end{equation}

\section*{Формулы коррекции витковой дальности при \(R\)-преобразовании II типа}

В разделе, посвящённом коррекции витковой дальности, используется линейная модель деформации полного угла. Скорректированная витковая дальность задаётся формулой
\begin{equation}
\lambda_{\mathrm{corr}}
=
\lambda
-
\frac{1}{2\pi}
\mathcal B_\lambda(\omega,\lambda,\kappa)\zeta.
\label{eq:appendixC_lambda_correction}
\end{equation}

Функция чувствительности \(\mathcal B_\lambda\) записывается в виде конечного гармонического разложения по долготе перицентра:
\begin{equation}
\begin{aligned}
\mathcal B_\lambda(\omega,\lambda,\kappa)
={}&
\mathcal B_0^{(\lambda)}(\lambda,\kappa)
+
\mathcal B_1^{(\lambda)}(\lambda,\kappa)\cos\omega
+
\mathcal B_2^{(\lambda)}(\lambda,\kappa)\sin\omega
\\
&+
\mathcal B_3^{(\lambda)}(\lambda,\kappa)\cos 3\omega
+
\mathcal B_4^{(\lambda)}(\lambda,\kappa)\sin 3\omega.
\end{aligned}
\label{eq:appendixC_B_lambda}
\end{equation}

Формулы для компонент \(\mathcal B_0^{(\lambda)},\ldots,\mathcal B_4^{(\lambda)}\) записываются с использованием обозначения
\begin{equation}
\varphi=2\pi\lambda.
\label{eq:appendixC_varphi_def}
\end{equation}

\begin{equation}
\mathcal B_0^{(\lambda)}(\lambda,\kappa)
=
\left(
\lambda\arctan(\kappa\lambda^2)+C_0^{(B_{\lambda,0})}
\right)
\left(
C_1^{(B_{\lambda,0})}\cos\left(C_2^{(B_{\lambda,0})}\lambda\right)
-C_3^{(B_{\lambda,0})}
-\frac{C_4^{(B_{\lambda,0})}}{\kappa}
\right)
\ln\kappa.
\label{eq:appendixC_B0_lam}
\end{equation}

\begin{equation}
\begin{aligned}
\mathcal B_1^{(\lambda)}(\lambda,\kappa)
={}&
\left[
\varphi+
\left(
-C_0^{(B_{\lambda,1})}\cos\varphi\,\arctan\kappa
-C_1^{(B_{\lambda,1})}
\right)
\right.
\\
&\left.
\cdot
\left(
\kappa\left(-\sqrt\varphi+\sin\varphi+C_2^{(B_{\lambda,1})}\right)
+\varphi-C_3^{(B_{\lambda,1})}
\right)
\right]
\ln\kappa.
\end{aligned}
\label{eq:appendixC_B1_lam}
\end{equation}

\begin{equation}
\begin{aligned}
\mathcal B_2^{(\lambda)}(\lambda,\kappa)
={}&
C_0^{(B_{\lambda,2})}
\left(C_1^{(B_{\lambda,2})}-\kappa\right)
\left[
\left(C_2^{(B_{\lambda,2})}\varphi+C_3^{(B_{\lambda,2})}\right)\sin\varphi
-\cos\varphi
+C_4^{(B_{\lambda,2})}
\right.
\\
&\left.
+
\frac{
\sin\left(
\varphi+\sqrt{\varphi\arctan(\arctan\kappa)}+C_5^{(B_{\lambda,2})}
\right)
-C_6^{(B_{\lambda,2})}
}{\kappa}
\right].
\end{aligned}
\label{eq:appendixC_B2_lam}
\end{equation}

\begin{equation}
\mathcal B_3^{(\lambda)}(\lambda,\kappa)
=
\varphi
\left[
-C_0^{(B_{\lambda,3})}
\left(
\kappa+
\sin\left(C_1^{(B_{\lambda,3})}\kappa\right)
\right)
\cos\left(
\varphi-
\frac{C_2^{(B_{\lambda,3})}}{\varphi}
\right)
-C_3^{(B_{\lambda,3})}
\right]
\ln\kappa
-
\left(
\frac{\kappa^2}{\varphi}
\right)^{C_4^{(B_{\lambda,3})}}.
\label{eq:appendixC_B3_lam}
\end{equation}

\begin{equation}
\begin{aligned}
\mathcal B_4^{(\lambda)}(\lambda,\kappa)
={}&
\frac{
\left(\kappa-C_0^{(B_{\lambda,4})}\right)
\left(\lambda+C_1^{(B_{\lambda,4})}\right)
}{C_2^{(B_{\lambda,4})}}
\left[
\sin\left(
\frac{\lambda}{C_3^{(B_{\lambda,4})}}
+C_4^{(B_{\lambda,4})}
\right)
\right.
\\
&\left.
+
\frac{
C_5^{(B_{\lambda,4})}
}{
\left(\lambda+C_6^{(B_{\lambda,4})}\right)^\kappa
+
\sin\left(
\frac{\lambda}{C_7^{(B_{\lambda,4})}}
\right)
}
\right].
\end{aligned}
\label{eq:appendixC_B4_lam}
\end{equation}
