\section{Линейные R-преобразования}

\begin{frame}
    \frametitle{Смысл линейного \(R\)-преобразования}
    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.42\textwidth}
            \small
            \[
            \mathbf p'=(R^T)^{-1}\mathbf p .
            \]
            \[
            \begin{array}{c}
            \mathrm{КЗ}_1
            \longrightarrow
            \mathrm{экстремаль}_1\\
            \xrightarrow{R}\\
            \mathrm{экстремаль}_2
            \longrightarrow
            \mathrm{КЗ}_2 .
            \end{array}
            \]
            \[
            \mathbf x_0=(\mathbf r_0,\mathbf v_0),
            \qquad
            \mathbf p_0=(\mathbf p_{r0},\mathbf p_{v0}).
            \]
            \begin{itemize}
                \item \(R\)-преобразование действует на начальные значения сопряжённых переменных.
                \item Оно не сохраняет оптимальность в той же задаче.
                \item Полученная траектория является экстремалью другой краевой задачи.
            \end{itemize}
        \end{column}
        \begin{column}{0.54\textwidth}
            \centering
            \includegraphics[width=\linewidth]{R_small_change_demo}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}[t]
    \frametitle{Параметры в блоке \(R\)-преобразований}
    \scriptsize
    \[
    \boldsymbol q=(\lambda,\kappa,\zeta,\chi,j,\gamma).
    \]
    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.48\textwidth}
            \begin{block}{Физические величины}
                \scriptsize
                \(e,\omega,i,\Omega,a_{\mathrm{orb}}\) -- целевые параметры конечной орбиты.
            \end{block}
            \begin{block}{Параметры матриц}
                \scriptsize
                \(\zeta,\chi,j,\gamma\) -- управляющие параметры \(R\)-преобразований.
            \end{block}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{block}{Служебные параметры}
                \scriptsize
                \(\varepsilon,\alpha,\beta\) -- внутренние параметры построения \(R_{e\omega}\).
            \end{block}
            \begin{block}{Композиция}
                \scriptsize
                \(R_0,R_{e\omega},R_{i\Omega},R_\Sigma\) -- последовательное преобразование начального вектора.
            \end{block}
        \end{column}
    \end{columns}

    \begin{block}{Как читать следующие слайды}
        \scriptsize
        Относительные углы \(L_0,\bar\omega_{\mathrm{rel}},\Omega_{\mathrm{rel}}\) переводят орбитальные долготы к углам матриц. Здесь \(a_{\mathrm{orb}}\) -- большая полуось конечной орбиты.
    \end{block}
\end{frame}

\begin{frame}[t]
    \frametitle{\(R\)-преобразование I типа}
    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.43\textwidth}
            \scriptsize
            \[
            R_\varphi=R(\mathbf r_0,\mathbf v_0(\varphi)).
            \]
            \[
            \mathbf p'_{r0}=(R_\varphi^T)^{-1}\mathbf p_{r0},
            \qquad
            \mathbf p'_{v0}=(R_\varphi^T)^{-1}\mathbf p_{v0}.
            \]
            \begin{itemize}
                \item \(R\)-I меняет начальный фазовый вектор.
                \item Начальный вектор сопряжённых переменных преобразуется согласованно с новым начальным фазовым вектором.
                \item Этот шаг используется далее как \(R_0\) в композиции.
            \end{itemize}
        \end{column}
        \begin{column}{0.53\textwidth}
            \centering
            \includegraphics[width=\linewidth,height=0.45\textheight,keepaspectratio]{R_type_I_demo}
        \end{column}
    \end{columns}
    \begin{newresult}
        Построено \(R\)-преобразование I типа для согласованного изменения начального фазового вектора и начального вектора сопряжённых переменных.
    \end{newresult}
\end{frame}

\begin{frame}[t]
    \frametitle{\(R\)-II: задание \(e\) и \(\omega\)}
    \tiny
    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.51\textwidth}
            \textbf{Матрица и поворот}
            \[
            \begin{gathered}
            w_1=-\sin\beta+\zeta,\quad w_2=\cos\beta,\\
            \varepsilon=w_1\cos\beta+w_2\sin\beta,\quad
            \alpha=-w_1\sin\beta+w_2\cos\beta.
            \end{gathered}
            \]
            \[
            \resizebox{0.92\linewidth}{!}{$
            R_e(\alpha,\varepsilon)
            =
            \begin{pmatrix}
            \alpha(1-\varepsilon) & 0 & 0\\
            0 & \dfrac{1+\varepsilon}{\sqrt{\alpha(1-\varepsilon^2)}} & 0\\
            0 & 0 &
            \dfrac{\alpha(1-\varepsilon)(1+\varepsilon)}{\sqrt{\alpha(1-\varepsilon^2)}}
            \end{pmatrix}.
            $}
            \]
            \[
            \begin{aligned}
            Q_\chi&=R_z(\chi),&
            R_{e\omega}&=Q_\chi R_e Q_\chi^T,\\
            \chi(\omega)&=\frac{\omega}{2}
            +\frac{1}{6}\sin 2\omega
            +\frac{1}{32}\sin 4\omega.
            \end{aligned}
            \]
            \begin{itemize}
                \item \(\zeta\) управляет деформацией формы орбиты.
                \item \(\chi\) задаёт поворот матрицы к целевому \(\omega\).
            \end{itemize}
        \end{column}
        \begin{column}{0.45\textwidth}
            \centering
            \includegraphics[width=0.70\linewidth,height=0.23\textheight,keepaspectratio]{R_type_II_Ce_isolines_raw}
            \textbf{Связь с целевым эксцентриситетом}
            \[
            e \approx -\mathcal A^{(e)}(\kappa,\omega)\zeta,
            \qquad
            \zeta=-\frac{e}{\mathcal A^{(e)}(\kappa,\omega)}.
            \]
            \[
            \resizebox{\linewidth}{!}{$
            \begin{aligned}
            \mathcal A^{(e)}(\kappa,\omega)
            &=
            \mathcal A_0^{(e)}(\kappa)
            +
            \sin\omega
            \left(
            \mathcal A_1^{(e)}(\kappa)\sin\omega
            +
            \mathcal A_2^{(e)}(\kappa)\sin3\omega
            \right),\\
            \mathcal A_m^{(e)}(\kappa)&=C_m^{(e)}(\kappa-1),
            \qquad m=0,1,2.
            \end{aligned}
            $}
            \]
            \begin{itemize}
                \item \(\mathcal A^{(e)}\) связывает управляющий параметр \(\zeta\) с целевым эксцентриситетом.
                \item Числовые коэффициенты функции не разворачиваются на слайде.
            \end{itemize}
        \end{column}
    \end{columns}
    \begin{newresult}
        Формулы \(R\)-II задают \(e\) и \(\omega\) конечной орбиты через параметры матрицы.
    \end{newresult}
\end{frame}

\begin{frame}[t]
    \frametitle{\(R\)-II: проверка и коррекция витковой дальности}
    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.48\textwidth}
            \centering
            \includegraphics[width=\linewidth,height=0.30\textheight,keepaspectratio]{R_type_II_eomega_example}

            \vspace{0.06cm}
            \includegraphics[width=0.82\linewidth,height=0.24\textheight,keepaspectratio]{R_type_II_lambda_correction_error_heatmap}
        \end{column}
        \begin{column}{0.48\textwidth}
            \tiny
            \[
            \begin{gathered}
            \Delta e=e_{\mathrm{num}}-e,\quad
            \Delta\omega=\operatorname{wrap}(\omega_{\mathrm{num}}-\omega),\\
            \Delta\lambda=\lambda_{\mathrm{num}}-\lambda,\quad
            \Delta a_{\mathrm{orb}}=a_{\mathrm{orb}}^{\mathrm{num}}-a_{\mathrm{orb}}.
            \end{gathered}
            \]
            \[
            \begin{gathered}
            \Delta\lambda
            \approx
            \frac{1}{2\pi}\mathcal B_\lambda(\omega,\lambda,\kappa)\zeta,\\
            \lambda_{\mathrm{corr}}
            =
            \lambda
            -
            \frac{1}{2\pi}\mathcal B_\lambda(\omega,\lambda,\kappa)\zeta.
            \end{gathered}
            \]
            \[
            \begin{aligned}
            \mathcal B_\lambda
            &=
            \mathcal B_0^{(\lambda)}
            +
            \mathcal B_1^{(\lambda)}\cos\omega
            +
            \mathcal B_2^{(\lambda)}\sin\omega\\
            &\quad+
            \mathcal B_3^{(\lambda)}\cos3\omega
            +
            \mathcal B_4^{(\lambda)}\sin3\omega .
            \end{aligned}
            \]
            \begin{itemize}
                \item Проверяются ошибки задания \(e,\omega\) и побочные деформации \(\lambda,a_{\mathrm{orb}}\).
                \item \(\mathcal B_\lambda\) -- функция чувствительности коррекции витковой дальности.
                \item Коррекция \(\lambda\) применяется до построения полной композиции.
            \end{itemize}
        \end{column}
    \end{columns}
    \begin{newresult}
        Проверка \(R\)-II дополнена линейной коррекцией побочной деформации витковой дальности.
    \end{newresult}
\end{frame}

\begin{frame}[t]
    \frametitle{\(R\)-преобразование III типа: задание \(i\) и \(\Omega\)}
    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.35\textwidth}
            \centering
            \includegraphics[width=\linewidth]{R_type_III_Ci_isolines}
        \end{column}
        \begin{column}{0.61\textwidth}
            \tiny
            \[
            R_i(j)=
            \begin{pmatrix}
            1 & 0 & 0\\
            0 & \dfrac{1}{1+j^2} & -\dfrac{j}{1+j^2}\\
            0 & \dfrac{j}{1+j^2} & \dfrac{1}{1+j^2}
            \end{pmatrix}.
            \]
            \[
            Q_\Omega=
            \begin{pmatrix}
            \cos\Omega & -\sin\Omega & 0\\
            \sin\Omega & \cos\Omega & 0\\
            0&0&1
            \end{pmatrix},
            \qquad
            R_{i\Omega}(j,\Omega)=Q_\Omega R_i(j)Q_\Omega^T.
            \]
            \[
            i\approx -\mathcal A^{(i)}(\kappa,\Omega)j,
            \qquad
            j=-\frac{i}{\mathcal A^{(i)}(\kappa,\Omega)}.
            \]
            \[
            \mathcal A^{(i)}(\kappa,\Omega)
            =
            \mathcal A_0^{(i)}(\kappa)
            +
            \mathcal A_1^{(i)}(\kappa)\cos 2\Omega.
            \]
            \[
            \mathcal A_0^{(i)}(\kappa)
            =
            \kappa
            \left(C_0^{(i)}+C_1^{(i)}\kappa\right)
            \left(\kappa+C_2^{(i)}\right)
            +
            C_3^{(i)},
            \]
            \[
            \mathcal A_1^{(i)}(\kappa)
            =
            C_4^{(i)}
            \left(\kappa-C_5^{(i)}\right)
            \left(\kappa^3-C_6^{(i)}\right).
            \]
            \begin{itemize}
                \item \(R\)-III выводит конечную орбиту из плоскости исходного движения.
                \item \(j\) -- управляющий параметр матрицы, связанный с наклонением.
                \item \(\Omega\) -- целевая долгота восходящего узла; \(\gamma\) в композиции задаёт соответствующий управляющий угол матрицы.
            \end{itemize}
        \end{column}
    \end{columns}
    \begin{newresult}
        Получены конечные формулы \(R\)-III для задания наклонения и долготы восходящего узла конечной орбиты.
    \end{newresult}
\end{frame}

\begin{frame}[t]
    \frametitle{Композиция \(R\)-преобразований}
    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.46\textwidth}
            \tiny
            \textbf{Относительные углы}
            \[
            \mathbf x_0=(\mathbf r_0,\mathbf v_0),
            \qquad
            R_0=R(\mathbf r_0,\mathbf v_0).
            \]
            \[
            \begin{aligned}
            \bar{\omega}&=\Omega+\omega,\\
            \bar{\omega}_{\mathrm{rel}}&=\bar{\omega}-L_0,\\
            \gamma&=\Omega_{\mathrm{rel}}=\Omega-L_0.
            \end{aligned}
            \]
            \textbf{Управляющие параметры и композиция}
            \[
            \zeta
            =
            -\frac{e}{\mathcal A^{(e)}(\kappa,\bar{\omega}_{\mathrm{rel}})}.
            \]
            \[
            j
            =
            -\frac{i}{\mathcal A^{(i)}(\kappa,\gamma)}.
            \]
            \[
            \begin{aligned}
            R_\Sigma
            &=
            R_{i\Omega}(j,\gamma)
            R_{e\omega}(\zeta,\bar{\omega}_{\mathrm{rel}})
            R_0.
            \end{aligned}
            \]
            \begin{itemize}
                \item \(R_0\) задаёт переход к начальному фазовому вектору.
                \item \(R_{e\omega}\) задаёт форму конечной орбиты.
                \item \(R_{i\Omega}\) задаёт пространственную ориентацию конечной орбиты.
            \end{itemize}
        \end{column}
        \begin{column}{0.50\textwidth}
            \centering
            \includegraphics[width=\linewidth,height=0.62\textheight,keepaspectratio]{R_composition_sequential_example}
        \end{column}
    \end{columns}
    \begin{newresult}
        Построена композиция \(R\)-преобразований для перехода к эллиптической и пространственной конечной орбите.
    \end{newresult}
\end{frame}

\begin{frame}
    \frametitle{Начальный вектор после композиции}
    \small
    \[
    \mathbf p_{r0}^{\mathrm{init}}
    =
    (R_{\Sigma}^{T})^{-1}
    \widehat{\mathbf p}_{r0}^{\mathrm{pl}},
    \qquad
    \mathbf p_{v0}^{\mathrm{init}}
    =
    (R_{\Sigma}^{T})^{-1}
    \widehat{\mathbf p}_{v0}^{\mathrm{pl}}.
    \]
    \[
    \mathbf y_0^{\mathrm{init}}
    =
    \left(
    \mathbf r_0,
    \mathbf v_0,
    \mathbf p_{r0}^{\mathrm{init}},
    \mathbf p_{v0}^{\mathrm{init}}
    \right).
    \]
    \begin{itemize}
        \item После композиции \(R_\Sigma\) управляющий вектор \(\boldsymbol q\) полностью задаёт начальный вектор расширенной системы.
        \item На входе используется аппроксимированный плоский начальный вектор сопряжённых переменных.
        \item Композиция \(R_\Sigma\) переносит его к заданному начальному фазовому вектору и конечной орбите.
        \item Полученный \(\mathbf y_0^{\mathrm{init}}\) используется как начальный вектор расширенной системы для последующего уточнения КЗ.
    \end{itemize}
\end{frame}
