\section{Линейные R-преобразования}

\begin{frame}
    \frametitle{Смысл линейного \(R\)-преобразования}
    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.42\textwidth}
            \small
            \[
            \mathbf p'=(R^T)^{-1}\mathbf p .
            \]
            \[
            \begin{array}{c}
            \mathrm{КЗ}_1
            \longrightarrow
            \mathrm{экстремаль}_1\\
            \xrightarrow{R}\\
            \mathrm{экстремаль}_2
            \longrightarrow
            \mathrm{КЗ}_2 .
            \end{array}
            \]
            \[
            \mathbf x_0=(\mathbf r_0,\mathbf v_0),
            \qquad
            \mathbf p_0=(\mathbf p_{r0},\mathbf p_{v0}).
            \]
            \begin{itemize}
                \item \(R\)-преобразование действует на начальные значения сопряжённых переменных.
                \item Полученная траектория является экстремалью другой краевой задачи.
            \end{itemize}
        \end{column}
        \begin{column}{0.54\textwidth}
            \centering
            \includegraphics[width=\linewidth]{R_small_change_demo}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}[t]
    \frametitle{\(R\)-преобразование I типа}
    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.43\textwidth}
            \scriptsize
            \[
            R_\varphi=R(\mathbf r_0,\mathbf v_0(\varphi)).
            \]
            \[
            \mathbf p'_{r0}=(R_\varphi^T)^{-1}\mathbf p_{r0},
            \qquad
            \mathbf p'_{v0}=(R_\varphi^T)^{-1}\mathbf p_{v0}.
            \]
            \begin{itemize}
                \item \(R\)-I меняет начальный фазовый вектор.
                \item Начальный вектор сопряжённых переменных преобразуется согласованно с новым начальным фазовым вектором.
                \item Этот шаг используется далее как \(R_0\) в композиции.
            \end{itemize}
        \end{column}

\begin{frame}
    \frametitle{\(R\)-преобразование II типа: задание \(e\) и \(\omega\)}
    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.36\textwidth}
            \centering
            \includegraphics[width=\linewidth]{R_type_II_Ce_isolines_raw}
        \end{column}
        \begin{column}{0.60\textwidth}
            \tiny
            \[
            \beta=0.642,\quad
            w_1=-\sin\beta+\zeta,\quad
            w_2=\cos\beta,
            \]
            \[
            \varepsilon=w_1\cos\beta+w_2\sin\beta,
            \qquad
            \alpha=-w_1\sin\beta+w_2\cos\beta.
            \]
            \[
            R_e(\alpha,\varepsilon)
            =
            \begin{pmatrix}
            \alpha(1-\varepsilon) & 0 & 0\\
            0 & \dfrac{1+\varepsilon}{\sqrt{\alpha(1-\varepsilon^2)}} & 0\\
            0 & 0 &
            \dfrac{\alpha(1-\varepsilon)(1+\varepsilon)}{\sqrt{\alpha(1-\varepsilon^2)}}
            \end{pmatrix}.
            \]
            \[
            Q_\chi=
            \begin{pmatrix}
            \cos\chi & -\sin\chi & 0\\
            \sin\chi & \cos\chi  & 0\\
            0        & 0         & 1
            \end{pmatrix},
            \qquad
            R_{e\omega}=Q_\chi R_e Q_\chi^T.
            \]
            \[
            \chi(\omega)
            =
            \frac{\omega}{2}
            +
            \frac{1}{6}\sin 2\omega
            +
            \frac{1}{32}\sin 4\omega.
            \]
            \[
            e \approx -\mathcal A^{(e)}(\kappa,\omega)\zeta,
            \qquad
            \zeta=-\frac{e}{\mathcal A^{(e)}(\kappa,\omega)}.
            \]
            \[
            \mathcal A^{(e)}(\kappa,\omega)
            =
            \mathcal A_0^{(e)}(\kappa)
            +
            \sin\omega
            \left(
            \mathcal A_1^{(e)}(\kappa)\sin\omega
            +
            \mathcal A_2^{(e)}(\kappa)\sin3\omega
            \right),
            \]
            \[
            \mathcal A_m^{(e)}(\kappa)=C_m^{(e)}(\kappa-1),
            \qquad
            m=0,1,2.
            \]
        \end{column}
    \end{columns}

\begin{frame}[t]
    \frametitle{\(R\)-преобразование II типа: проверка и побочная деформация}
    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.54\textwidth}
            \centering
            \includegraphics[height=0.54\textheight]{R_type_II_eomega_example}
        \end{column}
        \begin{column}{0.42\textwidth}
            \scriptsize
            \[
            \Delta e=e_{\mathrm{num}}-e,
            \qquad
            \Delta\omega=
            \operatorname{wrap}(\omega_{\mathrm{num}}-\omega).
            \]
            \[
            \Delta\lambda=\lambda_{\mathrm{num}}-\lambda,
            \qquad
            \Delta a=a_{\mathrm{num}}-a.
            \]
            \begin{itemize}
                \item Ошибка задания \(\omega\) оценивается как угловая ошибка на окружности.
                \item Окрестность малых эксцентриситетов исключается, поскольку при \(e=0\) аргумент перицентра вырожден.
                \item Преобразование вызывает побочную деформацию \(\lambda\) и \(a\).
                \item Для витковой дальности строится отдельная линейная коррекция.
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}

            \scriptsize
            \[
            \Delta\lambda
            \approx
            \frac{1}{2\pi}
            \mathcal B_\lambda(\omega,\lambda,\kappa)\zeta.
            \]
            \[
            \begin{aligned}
            \mathcal B_\lambda(\omega,\lambda,\kappa)
            &=
            \mathcal B_0^{(\lambda)}(\lambda,\kappa)
            +
            \mathcal B_1^{(\lambda)}(\lambda,\kappa)\cos\omega\\
            &\quad+
            \mathcal B_2^{(\lambda)}(\lambda,\kappa)\sin\omega
            +
            \mathcal B_3^{(\lambda)}(\lambda,\kappa)\cos 3\omega
            +
            \mathcal B_4^{(\lambda)}(\lambda,\kappa)\sin 3\omega .
            \end{aligned}
            \]
            \[
            \lambda_{\mathrm{corr}}
            =
            \lambda-\Delta\lambda
            =
            \lambda
            -
            \frac{1}{2\pi}
            \mathcal B_\lambda(\omega,\lambda,\kappa)\zeta.
            \]
            \begin{itemize}
                \item Локальная зависимость \(\Delta\lambda(\zeta)\) используется для выделения функции чувствительности.
                \item Коррекция применяется предварительно, до построения полной композиции преобразований.
            \end{itemize}

\begin{frame}
    \frametitle{\(R\)-преобразование III типа: задание \(i\) и \(\Omega\)}
    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.35\textwidth}
            \centering
            \includegraphics[width=\linewidth]{R_type_III_Ci_isolines}
        \end{column}
        \begin{column}{0.61\textwidth}
            \tiny
            \[
            R_i(j)=
            \begin{pmatrix}
            1 & 0 & 0\\
            0 & \dfrac{1}{1+j^2} & -\dfrac{j}{1+j^2}\\
            0 & \dfrac{j}{1+j^2} & \dfrac{1}{1+j^2}
            \end{pmatrix}.
            \]
            \[
            Q_\Omega=
            \begin{pmatrix}
            \cos\Omega & -\sin\Omega & 0\\
            \sin\Omega & \cos\Omega & 0\\
            0&0&1
            \end{pmatrix},
            \qquad
            R_{i\Omega}(j,\Omega)=Q_\Omega R_i(j)Q_\Omega^T.
            \]
            \[
            i\approx -\mathcal A^{(i)}(\kappa,\Omega)j,
            \qquad
            j=-\frac{i}{\mathcal A^{(i)}(\kappa,\Omega)}.
            \]
            \[
            \mathcal A^{(i)}(\kappa,\Omega)
            =
            \mathcal A_0^{(i)}(\kappa)
            +
            \mathcal A_1^{(i)}(\kappa)\cos 2\Omega.
            \]
            \[
            \mathcal A_0^{(i)}(\kappa)
            =
            \kappa
            \left(C_0^{(i)}+C_1^{(i)}\kappa\right)
            \left(\kappa+C_2^{(i)}\right)
            +
            C_3^{(i)},
            \]
            \[
            \mathcal A_1^{(i)}(\kappa)
            =
            C_4^{(i)}
            \left(\kappa-C_5^{(i)}\right)
            \left(\kappa^3-C_6^{(i)}\right).
            \]
            \begin{itemize}
                \item \(R\)-III выводит конечную орбиту из плоскости исходного движения.
                \item Управляющий параметр \(j\) задаёт наклонение, а поворот матрицы задаёт долготу восходящего узла.
            \end{itemize}
        \end{column}
    \end{columns}

\begin{frame}
    \frametitle{Композиция \(R\)-преобразований}
    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.46\textwidth}
            \tiny
            \[
            \mathbf x_0=(\mathbf r_0,\mathbf v_0),
            \qquad
            R_0=R(\mathbf r_0,\mathbf v_0).
            \]
            \[
            \bar{\omega}=\Omega+\omega,
            \qquad
            \bar{\omega}_{\mathrm{rel}}
            =
            \bar{\omega}-L_0,
            \qquad
            \Omega_{\mathrm{rel}}
            =
            \Omega-L_0.
            \]
            \[
            \gamma=\Omega_{\mathrm{rel}}.
            \]
            \[
            \zeta
            =
            -\frac{e}{\mathcal A^{(e)}(\kappa,\bar{\omega}_{\mathrm{rel}})},
            \qquad
            j
            =
            -\frac{i}{\mathcal A^{(i)}(\kappa,\gamma)}.
            \]
            \[
            R_\Sigma
            =
            R_{i\Omega}(j,\gamma)
            R_{e\omega}(\zeta,\bar{\omega}_{\mathrm{rel}})
            R_0.
            \]
            \begin{itemize}
                \item \(R_0\) задаёт переход к начальному фазовому вектору.
                \item \(R_{e\omega}\) задаёт форму конечной орбиты.
                \item \(R_{i\Omega}\) задаёт пространственную ориентацию конечной орбиты.
            \end{itemize}
        \end{column}
        \begin{column}{0.50\textwidth}
            \centering
            \includegraphics[height=0.62\textheight]{R_composition_sequential_example}
        \end{column}

\begin{frame}
    \frametitle{Начальный вектор после композиции}
    \small
    \[
    \mathbf p_{r0}^{\mathrm{init}}
    =
    (R_{\Sigma}^{T})^{-1}
    \widehat{\mathbf p}_{r0}^{\mathrm{pl}},
    \qquad
    \mathbf p_{v0}^{\mathrm{init}}
    =
    (R_{\Sigma}^{T})^{-1}
    \widehat{\mathbf p}_{v0}^{\mathrm{pl}}.
    \]
    \[
    \mathbf y_0^{\mathrm{init}}
    =
    \left(
    \mathbf r_0,
    \mathbf v_0,
    \mathbf p_{r0}^{\mathrm{init}},
    \mathbf p_{v0}^{\mathrm{init}}
    \right).
    \]
    \begin{itemize}
        \item На входе используется аппроксимированный плоский начальный вектор сопряжённых переменных.
        \item Композиция \(R_\Sigma\) переносит его к заданному начальному фазовому вектору и конечной орбите.
        \item Полученный \(\mathbf y_0^{\mathrm{init}}\) используется как начальный вектор расширенной системы для последующего уточнения КЗ.
    \end{itemize}
\end{frame}
