\section{Линейные R-преобразования}

\begin{frame}
    \frametitle{\(R\)-преобразование}
        \presentationsectionmark{Глава 3}
    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.42\textwidth}
            \presentationtextsize
            Матрица линейного преобразования  
            \vspace{-0.18cm}
            \[
            R(\boldsymbol\rho,\boldsymbol\nu)
            =
            \begin{pmatrix}
                \rho_x & \nu_x & \eta_x\\
                \rho_y & \nu_y & \eta_y\\
                \rho_z & \nu_z & \eta_z
            \end{pmatrix}.
            \]
            \[
            \boldsymbol\eta=\boldsymbol\rho\times\boldsymbol\nu .
            \]            
            \[
            \mathbf p_0=(\mathbf p_{r0},\mathbf p_{v0}).
            \]
            \[
            \mathbf p_0'=(R^T)^{-1}\mathbf p_0 .
            \]
            \vspace{-0.18cm}
            \begin{itemize}
                \item \(R\)-преобразование действует на начальные значения сопряжённых переменных
                \item Затем интегрируется расширенная система уравнений движения 
                \item После интегрирования получается экстремаль новой краевой задачи, не изначальной
            \end{itemize}
        \end{column}
        \begin{column}{0.57\textwidth}
            \presentationtextsize
            \centering
            \includegraphics[width=\linewidth]{R_small_change_demo}
            Пример действия \(R\)-преобразования на траекторию
            \vspace{-0.18cm}
            \[
            R=
            \begin{pmatrix}
                1.08 & 0 & 0\\
                0    & 1 & 0\\
                0    & 0 & 1.08
            \end{pmatrix}.
            \]
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}[t]
    \frametitle{Инварианты и допустимый вид матрицы \(R\)}
        \presentationsectionmark{Глава 3}
    \presentationtextsize
    Первый интеграл расширенной системы
    \[
    \mathbf M=\mathbf r\times\mathbf p_r+\mathbf v\times\mathbf p_v
    \]
    Нормаль к исходной плоскости
    \[
    \mathbf c_0=\mathbf r_0\times\mathbf v_0=(0,0,1)^T 
    \]
    Скалярный и векторный инварианты
    \[
    I_0=\mathbf c_0\cdot\mathbf M,\quad
    \mathbf I_1=\mathbf c_0\times\mathbf M 
    \]
    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.48\textwidth}
            Плоские преобразования
            \[
            \mathbf I'_1=\mathbf 0
            \quad\Rightarrow\quad
            R_e=
            \begin{pmatrix}
            \rho_x & \nu_x & 0\\
            \rho_y & \nu_y & 0\\
            0&0&\eta_z
            \end{pmatrix}.
            \]
        \end{column}
        \begin{column}{0.48\textwidth}
            Пространственные преобразования
            \[
            I'_0=I_0
            \quad\Rightarrow\quad
            R_i(j)=
            \begin{pmatrix}
            1&0&0\\
            0&\dfrac{1}{1+j^2}&-\dfrac{j}{1+j^2}\\
            0&\dfrac{j}{1+j^2}&\dfrac{1}{1+j^2}
            \end{pmatrix}.
            \]
        \end{column}
    \end{columns}
    \begin{newresult}
        С помощью инвариантов записаны условия на допустимый вид матрицы \(R\) в плоском и в пространственном случаях
    \end{newresult}
\end{frame}

\begin{frame}[t]
    \frametitle{\(R\)-преобразование I типа}
        \presentationsectionmark{Глава 3}
    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.42\textwidth}
            \presentationtextsize
            Составляем матрицу \(R\) из радиус-вектора, скорости и орбитального момента
            \vspace{-0.18cm}
            \[
            R_0=R(\mathbf r'_0,\mathbf v'_0).
            \]
            \[
            \begin{aligned}
            \mathbf r'_0 &= R_0\mathbf r_0,\\
            \mathbf v'_0 &= R_0\mathbf r_0,\\
            \mathbf p'_{r0} &= (R_0^T)^{-1}\mathbf p_{r0},\\
            \mathbf p'_{v0} &= (R_0^T)^{-1}\mathbf p_{v0}.
            \end{aligned}
            \]
            \begin{itemize}
                \item \(R\)-I изменяет начальный сопряжённый вектор согласованно с фазовым вектором
                \item В сравнении с простым поворотом невязки правого конца уменьшаются на \(2.5\) порядка
            \end{itemize}
        \end{column}
        \begin{column}{0.57\textwidth}
            \presentationtextsize
            \centering
            \includegraphics[width=\linewidth]{R_type_I_demo}
            Получается другая траектория с той же матрицей \(R\)
            \vspace{-0.18cm}
            \[
            R=
            \begin{pmatrix}
                1.08 & 0 & 0\\
                0    & 1 & 0\\
                0    & 0 & 1.08
            \end{pmatrix}.
            \]
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}[t]
    \frametitle{\(R\)-преобразование II типа}
        \presentationsectionmark{Глава 3}
    \presentationtextsize
    \vspace{-0.18cm}
    Матрица плоского преобразования
    \vspace{-0.18cm}
    \[
    R_e(\alpha,\varepsilon)
    =
    \begin{pmatrix}
    \alpha(1-\varepsilon) & 0 & 0\\
    0 & \dfrac{1+\varepsilon}{\sqrt{\alpha(1-\varepsilon^2)}} & 0\\
    0 & 0 &
    \dfrac{\alpha(1-\varepsilon)(1+\varepsilon)}{\sqrt{\alpha(1-\varepsilon^2)}}
    \end{pmatrix}.
    \]
    \vspace{-0.18cm}
    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.48\textwidth}
            Промежуточная репараметризация
            \vspace{-0.18cm}
            \[
            \makebox[\linewidth][l]{$\displaystyle
            \begin{aligned}
            w_1&=-\sin\beta+\zeta,&
            w_2&=\cos\beta,\\
            \varepsilon&=w_1\cos\beta+w_2\sin\beta,&
            \alpha&=-w_1\sin\beta+w_2\cos\beta.
            \end{aligned}
            $}
            \]
            Поворот подобными матрицами
            \vspace{-0.18cm}
            \[
            \makebox[\linewidth][l]{$\displaystyle
            Q_\chi=
            \begin{pmatrix}
            \cos\chi & -\sin\chi & 0\\
            \sin\chi & \cos\chi & 0\\
            0&0&1
            \end{pmatrix}.
            $}
            \]
            \vspace{-0.18cm}
            \[
            \makebox[\linewidth][l]{$\displaystyle
            \begin{aligned}
            R_{e\omega}&=Q_\chi R_e Q_\chi^T,\\
            \chi(\omega)&=\frac{\omega}{2}
            +\frac{1}{6}\sin 2\omega
            +\frac{1}{32}\sin 4\omega.
            \end{aligned}
            $}
            \]
            Аппроксимация эксцентриситета
            \vspace{-0.18cm}
            \[
            \makebox[\linewidth][l]{$\displaystyle
            e \approx -\mathcal A^{(e)}(\kappa,\omega)\zeta,
            \qquad
            \zeta=-\frac{e}{\mathcal A^{(e)}(\kappa,\omega)}.
            $}
            \]
        \end{column}
        \begin{column}{0.48\textwidth}
            \centering
            \includegraphics[width=\linewidth,height=0.345\textheight,keepaspectratio]{R_type_II_Ce_isolines_raw}

            \vspace{-0.02cm}
            \raggedright
            Аппроксимация коэффициент-функции
            \vspace{-0.18cm}
            \[
            \makebox[\linewidth][l]{$\displaystyle
            \begin{aligned}
            \mathcal A^{(e)}(\kappa,\omega)
            &=
            \mathcal A_0^{(e)}(\kappa)
            +
            \sin\omega
            \Bigl(\\[-0.2em]
            &\quad
            \mathcal A_1^{(e)}(\kappa)\sin\omega
            +
            \mathcal A_2^{(e)}(\kappa)\sin3\omega
            \Bigr),\\
            \mathcal A_m^{(e)}(\kappa)&=C_m^{(e)}(\kappa-1),
            \qquad m=0,1,2.
            \end{aligned}
            $}
            \]
        \end{column}
    \end{columns}
\end{frame}

            \presentationtextsize
            Коррекция невязки
            \[
            \begin{aligned}
            \Delta\lambda
            &\approx
            \frac{1}{2\pi}\mathcal B_\lambda(\omega,\lambda,\kappa)\zeta,\\
            \lambda_{\mathrm{corr}}
            &=
            \lambda
            -
            \frac{1}{2\pi}\mathcal B_\lambda(\omega,\lambda,\kappa)\zeta.
            \end{aligned}
            \]
            Коэффициент-функция \(\mathcal B\)
            \[
            \begin{aligned}
            \mathcal B_\lambda
            &=
            \mathcal B_0^{(\lambda)}
            +
            \mathcal B_1^{(\lambda)}\cos\omega
            +
            \mathcal B_2^{(\lambda)}\sin\omega\\
            &\quad+
            \mathcal B_3^{(\lambda)}\cos3\omega
            +
            \mathcal B_4^{(\lambda)}\sin3\omega .
            \end{aligned}
            \]

        \end{column}
    \end{columns}
\end{frame}

            1 & 0 & 0\\

\begin{frame}[t]
    \frametitle{Композиция \(R\)-преобразований}
        \presentationsectionmark{Глава 3}
    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.42\textwidth}
            \presentationtextsize
            
            \[
            \begin{aligned}
            R_\Sigma
            &=
            R_{i\Omega}(j,\gamma)
            R_{e\omega}(\zeta,\chi)
            R_0.
            \end{aligned}
            \]
            \begin{itemize}
                \item \(R_0\) задаёт начальную конфигурацию
                \item \(R_{e\omega}\) задаёт форму конечной орбиты
                \item \(R_{i\Omega}\) задаёт пространственную ориентацию конечной орбиты
            \end{itemize}
        \end{column}
        \begin{column}{0.57\textwidth}
            \centering
            \includegraphics[width=\linewidth]{R_composition_sequential_example}
        \end{column}
    \end{columns}
    \begin{newresult}
        Предложена композиция трёх линейных \(R\)-преобразований для перехода от компланарных перелётов между круговыми орбитами к некомпланарным перелётам между эллиптическими орбитами
    \end{newresult}
\end{frame}
