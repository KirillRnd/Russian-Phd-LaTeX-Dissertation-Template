\section{Линейные R-преобразования}

\begin{frame}
    \frametitle{Смысл линейного \(R\)-преобразования}
    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.42\textwidth}
            \presentationtextsize
            \[
            \mathbf p'=(R^T)^{-1}\mathbf p .
            \]
            \[
            \begin{array}{c}
            \mathrm{КЗ}_1
            \longrightarrow
            \mathrm{экстремаль}_1\\
            \xrightarrow{R}\\
            \mathrm{экстремаль}_2
            \longrightarrow
            \mathrm{КЗ}_2 .
            \end{array}
            \]
            \[
            \mathbf x_0=(\mathbf r_0,\mathbf v_0),
            \qquad
            \mathbf p_0=(\mathbf p_{r0},\mathbf p_{v0}).
            \]
            \begin{itemize}
                \item Мы не деформируем готовую траекторию.
                \item Меняются начальные значения сопряжённых переменных, затем расширенная система интегрируется заново.
                \item Полученная траектория является экстремалью другой краевой задачи, а не доказанным глобальным оптимумом.
            \end{itemize}
        \end{column}
        \begin{column}{0.54\textwidth}
            \centering
            \includegraphics[width=\linewidth]{R_small_change_demo}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}[t]
    \frametitle{Инварианты и допустимый вид матрицы \(R\)}
    \presentationcompactsize
    \[
    \mathbf M=\mathbf r\times\mathbf p_r+\mathbf v\times\mathbf p_v,
    \qquad
    I_0=\mathbf c_0\cdot\mathbf M,\quad
    \mathbf I_1=\mathbf c_0\times\mathbf M .
    \]
    \[
    \mathbf p'_{r0}=A\mathbf p_{r0},\qquad
    \mathbf p'_{v0}=A\mathbf p_{v0},\qquad
    A=(R^T)^{-1}.
    \]
    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.48\textwidth}
            \textbf{Плоский класс}
            \[
            \mathbf I'_1=\mathbf 0
            \quad\Rightarrow\quad
            R_e=
            \begin{pmatrix}
            \rho_x & \nu_x & 0\\
            \rho_y & \nu_y & 0\\
            0&0&\eta_z
            \end{pmatrix}.
            \]
            Основа \(R\)-II для задания \(e,\omega\).
        \end{column}
        \begin{column}{0.48\textwidth}
            \textbf{Пространственный класс}
            \[
            I'_0=I_0
            \quad\Rightarrow\quad
            R_i(j)=
            \begin{pmatrix}
            1&0&0\\
            0&\dfrac{1}{1+j^2}&-\dfrac{j}{1+j^2}\\
            0&\dfrac{j}{1+j^2}&\dfrac{1}{1+j^2}
            \end{pmatrix}.
            \]
            Основа \(R\)-III для задания \(i,\Omega\).
        \end{column}
    \end{columns}
    \begin{newresult}
        Инварианты задают класс матриц; конечные формулы затем выбирают их параметры.
    \end{newresult}
\end{frame}

\begin{frame}[t]
    \frametitle{Параметры в блоке \(R\)-преобразований}
    \presentationtextsize
    \[
    \boldsymbol q=(\lambda,\kappa,\zeta,\chi,j,\gamma).
    \]
    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.48\textwidth}
            \begin{block}{Физические величины}
                \presentationtextsize
                \(e,\omega,i,\Omega,a_{\mathrm{orb}}\) -- целевые параметры конечной орбиты.
            \end{block}
            \begin{block}{Параметры матриц}
                \presentationtextsize
                \(\zeta,\chi,j,\gamma\) -- управляющие параметры \(R\)-преобразований.
            \end{block}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{block}{Служебные параметры}
                \presentationtextsize
                \(\varepsilon,\alpha,\beta\) -- внутренние параметры построения \(R_{e\omega}\).
            \end{block}
            \begin{block}{Композиция}
                \presentationtextsize
                \(R_0,R_{e\omega},R_{i\Omega},R_\Sigma\) -- последовательное преобразование начального вектора.
            \end{block}
        \end{column}
    \end{columns}

    \begin{block}{Как читать следующие слайды}
        \presentationcompactsize
        Вектор \(\boldsymbol q\) неявно задаёт постановку КЗ: от плоской модельной ветви выполняется переход к заданному начальному фазовому вектору и конечной орбите.
    \end{block}
\end{frame}

\begin{frame}[t]
    \frametitle{\(R\)-преобразование I типа}
    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.43\textwidth}
            \presentationtextsize
            \[
            R_\varphi=R(\mathbf r_0,\mathbf v_0(\varphi)).
            \]
            \[
            \mathbf p'_{r0}=(R_\varphi^T)^{-1}\mathbf p_{r0},
            \qquad
            \mathbf p'_{v0}=(R_\varphi^T)^{-1}\mathbf p_{v0}.
            \]
            \begin{itemize}
                \item \(R\)-I меняет начальный фазовый вектор и не сводится к простому повороту.
                \item Его роль -- учесть неортогональность \(\mathbf r_0\) и \(\mathbf v_0\) на эллиптической начальной орбите.
                \item В сравнении с простым поворотом невязки правого конца уменьшаются примерно на \(2.5\) порядка.
                \item Этот шаг используется далее как \(R_0\) в композиции.
            \end{itemize}
        \end{column}
        \begin{column}{0.53\textwidth}
            \centering
            \includegraphics[width=\linewidth,height=0.45\textheight,keepaspectratio]{R_type_I_demo}
        \end{column}
    \end{columns}
    \begin{newresult}
        \(R\)-I согласованно меняет начальную конфигурацию и сопряжённые переменные.
    \end{newresult}
\end{frame}

\begin{frame}[t]
    \frametitle{\(R\)-II: задание \(e\) и \(\omega\)}
    \presentationcompactsize
    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.51\textwidth}
            \textbf{Матрица и поворот}
            \[
            \begin{gathered}
            w_1=-\sin\beta+\zeta,\quad w_2=\cos\beta,\\
            \varepsilon=w_1\cos\beta+w_2\sin\beta,\quad
            \alpha=-w_1\sin\beta+w_2\cos\beta.
            \end{gathered}
            \]
            \[
            \resizebox{0.92\linewidth}{!}{$
            R_e(\alpha,\varepsilon)
            =
            \begin{pmatrix}
            \alpha(1-\varepsilon) & 0 & 0\\
            0 & \dfrac{1+\varepsilon}{\sqrt{\alpha(1-\varepsilon^2)}} & 0\\
            0 & 0 &
            \dfrac{\alpha(1-\varepsilon)(1+\varepsilon)}{\sqrt{\alpha(1-\varepsilon^2)}}
            \end{pmatrix}.
            $}
            \]
            \[
            \begin{aligned}
            Q_\chi&=R_z(\chi),&
            R_{e\omega}&=Q_\chi R_e Q_\chi^T,\\
            \chi(\omega)&=\frac{\omega}{2}
            +\frac{1}{6}\sin 2\omega
            +\frac{1}{32}\sin 4\omega.
            \end{aligned}
            \]
            \begin{itemize}
                \item \(\beta\) выделяет среднее по \(\kappa\) направление максимального изменения эксцентриситета.
                \item \(\zeta\) задаёт смещение вдоль этого направления.
                \item \(\chi\) -- вспомогательный угол поворота матрицы к целевому \(\omega\).
            \end{itemize}
        \end{column}
        \begin{column}{0.45\textwidth}
            \centering
            \includegraphics[width=0.70\linewidth,height=0.23\textheight,keepaspectratio]{R_type_II_Ce_isolines_raw}
            \textbf{Связь с целевым эксцентриситетом}
            \[
            e \approx -\mathcal A^{(e)}(\kappa,\omega)\zeta,
            \qquad
            \zeta=-\frac{e}{\mathcal A^{(e)}(\kappa,\omega)}.
            \]
            \[
            \resizebox{\linewidth}{!}{$
            \begin{aligned}
            \mathcal A^{(e)}(\kappa,\omega)
            &=
            \mathcal A_0^{(e)}(\kappa)
            +
            \sin\omega
            \left(
            \mathcal A_1^{(e)}(\kappa)\sin\omega
            +
            \mathcal A_2^{(e)}(\kappa)\sin3\omega
            \right),\\
            \mathcal A_m^{(e)}(\kappa)&=C_m^{(e)}(\kappa-1),
            \qquad m=0,1,2.
            \end{aligned}
            $}
            \]
            \begin{itemize}
                \item Аналитика задаёт форму \(R_{e\omega}\).
                \item Конечная формула \(\mathcal A^{(e)}\) выбирает параметр \(\zeta\) под заданный \(e\).
            \end{itemize}
        \end{column}
    \end{columns}
    \begin{newresult}
        \(R\)-II задаёт форму конечной орбиты через \(e\) и \(\omega\).
    \end{newresult}
\end{frame}

\begin{frame}[t]
    \frametitle{\(R\)-II: проверка и коррекция витковой дальности}
    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.48\textwidth}
            \centering
            \includegraphics[width=\linewidth,height=0.30\textheight,keepaspectratio]{R_type_II_eomega_example}

            \vspace{0.06cm}
            \includegraphics[width=0.82\linewidth,height=0.24\textheight,keepaspectratio]{R_type_II_lambda_correction_error_heatmap}
        \end{column}
        \begin{column}{0.48\textwidth}
            \presentationcompactsize
            \[
            \begin{gathered}
            \Delta e=e_{\mathrm{num}}-e,\quad
            \Delta\omega=\operatorname{wrap}(\omega_{\mathrm{num}}-\omega),\\
            \Delta\lambda=\lambda_{\mathrm{num}}-\lambda,\quad
            \Delta a_{\mathrm{orb}}=a_{\mathrm{orb}}^{\mathrm{num}}-a_{\mathrm{orb}}.
            \end{gathered}
            \]
            \[
            \begin{gathered}
            \Delta\lambda
            \approx
            \frac{1}{2\pi}\mathcal B_\lambda(\omega,\lambda,\kappa)\zeta,\\
            \lambda_{\mathrm{corr}}
            =
            \lambda
            -
            \frac{1}{2\pi}\mathcal B_\lambda(\omega,\lambda,\kappa)\zeta.
            \end{gathered}
            \]
            \[
            \begin{aligned}
            \mathcal B_\lambda
            &=
            \mathcal B_0^{(\lambda)}
            +
            \mathcal B_1^{(\lambda)}\cos\omega
            +
            \mathcal B_2^{(\lambda)}\sin\omega\\
            &\quad+
            \mathcal B_3^{(\lambda)}\cos3\omega
            +
            \mathcal B_4^{(\lambda)}\sin3\omega .
            \end{aligned}
            \]
            \begin{itemize}
                \item \(R\)-II задаёт \(e,\omega\), но ожидаемо деформирует \(\lambda\) в нелинейной динамике.
                \item Коррекция применяется к \(\lambda\), поскольку она связана с выбором ветви и продолжительностью перелёта.
                \item Одновременная простая коррекция \(\lambda,\kappa\) не используется из-за нелинейного взаимодействия поправок.
            \end{itemize}
        \end{column}
    \end{columns}
    \begin{newresult}
        Коррекция \(\lambda\) компенсирует ожидаемую побочную деформацию \(R\)-II.
    \end{newresult}
\end{frame}

\begin{frame}[t]
    \frametitle{\(R\)-преобразование III типа: задание \(i\) и \(\Omega\)}
    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.35\textwidth}
            \centering
            \includegraphics[width=\linewidth]{R_type_III_Ci_isolines}
        \end{column}
        \begin{column}{0.61\textwidth}
            \presentationcompactsize
            \[
            R_i(j)=
            \begin{pmatrix}
            1 & 0 & 0\\
            0 & \dfrac{1}{1+j^2} & -\dfrac{j}{1+j^2}\\
            0 & \dfrac{j}{1+j^2} & \dfrac{1}{1+j^2}
            \end{pmatrix}.
            \]
            \[
            Q_\Omega=
            \begin{pmatrix}
            \cos\Omega & -\sin\Omega & 0\\
            \sin\Omega & \cos\Omega & 0\\
            0&0&1
            \end{pmatrix},
            \qquad
            R_{i\Omega}(j,\Omega)=Q_\Omega R_i(j)Q_\Omega^T.
            \]
            \[
            i\approx -\mathcal A^{(i)}(\kappa,\Omega)j,
            \qquad
            j=-\frac{i}{\mathcal A^{(i)}(\kappa,\Omega)}.
            \]
            \[
            \mathcal A^{(i)}(\kappa,\Omega)
            =
            \mathcal A_0^{(i)}(\kappa)
            +
            \mathcal A_1^{(i)}(\kappa)\cos 2\Omega.
            \]
            \[
            \mathcal A_0^{(i)}(\kappa)
            =
            \kappa
            \left(C_0^{(i)}+C_1^{(i)}\kappa\right)
            \left(\kappa+C_2^{(i)}\right)
            +
            C_3^{(i)},
            \]
            \[
            \mathcal A_1^{(i)}(\kappa)
            =
            C_4^{(i)}
            \left(\kappa-C_5^{(i)}\right)
            \left(\kappa^3-C_6^{(i)}\right).
            \]
            \begin{itemize}
                \item \(R\)-III выводит конечную орбиту из плоскости исходного движения.
                \item \(j,\gamma\) -- управляющие параметры матрицы.
                \item \(i,\Omega\) -- целевые орбитальные элементы.
                \item Это тот же принцип деформации сопряжённых переменных, но для пространственной ориентации конечной орбиты.
            \end{itemize}
        \end{column}
    \end{columns}
    \begin{newresult}
        \(R\)-III задаёт пространственную ориентацию конечной орбиты.
    \end{newresult}
\end{frame}

\begin{frame}
    \frametitle{\(R\)-преобразование III типа: проверка и побочные деформации}

            \centering
            \includegraphics[width=\linewidth]{R_type_III_iOmega_example}

            \presentationtextsize
            \[
            \Delta i=i_{\mathrm{num}}-i,
            \]
            \[
            \Delta\Omega=
            \operatorname{wrap}(\Omega_{\mathrm{num}}-\Omega).
            \]
            \begin{itemize}
                \item Ошибка задания \(\Omega\) оценивается как угловая ошибка на окружности.
                \item Окрестность малых наклонений исключается, поскольку при \(i=0\) долгота восходящего узла вырождена.
                \item Побочные деформации \(\lambda\) и \(a_{\mathrm{orb}}\) контролируются отдельно от ошибок задания \(i,\Omega\).
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}[t]
    \frametitle{Композиция \(R\)-преобразований}
    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.46\textwidth}
            \presentationcompactsize
            \textbf{Относительные углы}
            \[
            \mathbf x_0=(\mathbf r_0,\mathbf v_0),
            \qquad
            R_0=R(\mathbf r_0,\mathbf v_0).
            \]
            \[
            \begin{aligned}
            \bar{\omega}&=\Omega+\omega,\\
            \bar{\omega}_{\mathrm{rel}}&=\bar{\omega}-L_0,\\
            \gamma&=\Omega_{\mathrm{rel}}=\Omega-L_0.
            \end{aligned}
            \]
            \textbf{Управляющие параметры и композиция}
            \[
            \zeta
            =
            -\frac{e}{\mathcal A^{(e)}(\kappa,\bar{\omega}_{\mathrm{rel}})}.
            \]
            \[
            j
            =
            -\frac{i}{\mathcal A^{(i)}(\kappa,\gamma)}.
            \]
            \[
            \begin{aligned}
            R_\Sigma
            &=
            R_{i\Omega}(j,\gamma)
            R_{e\omega}(\zeta,\bar{\omega}_{\mathrm{rel}})
            R_0.
            \end{aligned}
            \]
            \begin{itemize}
                \item \(R_0\) задаёт начальную конфигурацию.
                \item \(R_{e\omega}\) задаёт форму конечной орбиты.
                \item \(R_{i\Omega}\) задаёт её пространственную ориентацию.
                \item Типы I--III являются частями одной композиции, а не тремя независимыми приёмами.
            \end{itemize}
        \end{column}
        \begin{column}{0.50\textwidth}
            \centering
            \includegraphics[width=\linewidth,height=0.62\textheight,keepaspectratio]{R_composition_sequential_example}
        \end{column}
    \end{columns}
    \begin{newresult}
        Композиция собирает переход от модельной ветви к целевой постановке.
    \end{newresult}
\end{frame}

\begin{frame}
    \frametitle{Начальные значения после композиции}
    \presentationtextsize
    \[
    \mathbf p_{r0}^{\mathrm{init}}
    =
    (R_{\Sigma}^{T})^{-1}
    \widehat{\mathbf p}_{r0}^{\mathrm{pl}},
    \qquad
    \mathbf p_{v0}^{\mathrm{init}}
    =
    (R_{\Sigma}^{T})^{-1}
    \widehat{\mathbf p}_{v0}^{\mathrm{pl}}.
    \]
    \[
    \mathbf y_0^{\mathrm{init}}
    =
    \left(
    \mathbf r_0,
    \mathbf v_0,
    \mathbf p_{r0}^{\mathrm{init}},
    \mathbf p_{v0}^{\mathrm{init}}
    \right).
    \]
    \begin{itemize}
        \item На входе используется \(\widehat{\mathbf z}^{\mathrm{pl}}_0\) из блока конечных формул.
        \item Композиция \(R_\Sigma\) переносит начальные значения сопряжённых переменных к заданной постановке.
        \item Вектор \(\boldsymbol q\) тем самым задаёт начальное приближение для расширенной системы.
        \item Полученный \(\mathbf y_0^{\mathrm{init}}\) используется для последующего уточнения КЗ.
    \end{itemize}
\end{frame}
