\section{Конечные формулы}

\begin{frame}[t]
    \frametitle{Обозначения конечных формул}
    \small
    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.48\textwidth}
            \begin{block}{Величины и аргументы}
                \scriptsize
                \begin{itemize}
                    \item \(\widehat{x}\) -- конечная формула для величины \(x\);
                    \item \(x^\star\) -- значение, полученное из численного решения краевой задачи;
                    \item \(\lambda^\star\) -- витковая дальность в точке оптимальной дальности;
                    \item \(\kappa\) -- отношение радиусов модельных круговых орбит.
                \end{itemize}
            \end{block}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{block}{Коэффициенты и функции}
                \scriptsize
                \begin{itemize}
                    \item \(C_i^{(X)}\) -- числовые коэффициенты формулы для величины \(X\);
                    \item \(\mathcal A^{(X)}\) -- коэффициентные функции;
                    \item \(\mathcal D_m^{(X)}\) -- поправочные функции в формулах управления;
                    \item \(\mathcal P_i^{(\kappa)}\) -- вспомогательные функции согласования по \(\kappa\);
                    \item \(\mathcal B_\lambda\) -- функция коррекции витковой дальности.
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}

    \begin{block}{Как читать следующие слайды}
        \scriptsize
        Формулы задают компактное описание основной области модельного многообразия; развёрнутые коэффициенты и служебные функции вынесены в приложения диссертации.
    \end{block}
\end{frame}

                \item \(\kappa\) -- отношение радиусов модельных круговых орбит.
                \item Формула задаёт продолжительность перелёта на основной области модельного многообразия.
            \end{itemize}
        \end{column}

    \end{newresult}

\begin{frame}[t]
    \frametitle{Конечные формулы для непрерывной фазы}
    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.40\textwidth}
            \centering
            \includegraphics[width=\linewidth,height=0.29\textheight,keepaspectratio]{main_phase_isolines}

            \vspace{0.06cm}
            \includegraphics[width=\linewidth,height=0.29\textheight,keepaspectratio]{main_phase_error_heatmap}
        \end{column}
        \begin{column}{0.56\textwidth}
            \tiny
            \[
            \Psi =
            \lambda^\star-\frac{T}{\kappa^{3/2}},
            \qquad
            \widehat{\Psi}_{T}
            =
            \lambda^\star-\frac{\widehat{T}}{\kappa^{3/2}}.
            \]
            \[
            \widehat{\Psi}_{T}\left(\lambda^\star,\kappa\right)=
            \left(
            1-\frac{1}{\sqrt{\kappa}}
            +\frac{\sqrt6}{9}\frac{\ln\kappa}{\sqrt{\kappa}}
            \right)\lambda^\star
            -
            \frac{2}{27}\frac{\ln\kappa}{\sqrt{\kappa}}.
            \]
            \[
            \widehat{\Psi}\left(\lambda^\star,\kappa\right)
            =
            \mathcal A_0^{(\Psi)}(\kappa)
            +
            \mathcal A_1^{(\Psi)}(\kappa)\lambda^\star .
            \]
            \[
            \begin{aligned}
            \mathcal A_0^{(\Psi)}(\kappa)
            &=
            -C_0^{(\Psi)} e^{-\kappa}\ln^2\kappa,\\
            \mathcal A_1^{(\Psi)}(\kappa)
            &=
            C_0^{(\Psi)}
            \left(e^\kappa+1\right)e^{-\kappa}\ln\kappa,
            \end{aligned}
            \]
            \begin{itemize}
                \item \(\Psi\) -- нормированная непрерывная фаза; \(\Phi\) остаётся начальной синодической фазой.
                \item \(\widehat{\Psi}_T\) наследует ошибку формулы для \(\widehat T\).
                \item \(\widehat{\Psi}\) строится как самостоятельная конечная формула.
            \end{itemize}
        \end{column}
    \end{columns}
    \begin{newresult}
        Конечные формулы непрерывной фазы.
    \end{newresult}
\end{frame}

                \item \(\widehat J\) задаёт значение квадратичного функционала на аппроксимированном многообразии.
                \item Формула используется для оценки энергетической стоимости модельного перелёта.
                \item Числовые коэффициенты формулы приведены в приложениях диссертации.

\begin{frame}[t]
    \frametitle{Конечные формулы для начального вектора: компонентные формулы}
    \tiny
    \vspace{-0.08cm}
    \[
    \widehat{\mathbf z}^{\mathrm{pl}}_0=
    \left(
    \widehat{p}_{rx},
    \widehat{p}_{ry},
    \widehat{p}_{vx},
    \widehat{p}_{vy}
    \right).
    \]
    \vspace{-0.08cm}
    \[
    \widehat{p}_{rx}\left(\lambda^\star\right)=
    \frac{
    C_0^{(p_{rx})}
    }{
    C_1^{(p_{rx})}\lambda^\star
    +
    C_2^{(p_{rx})}
    +
    \sin\left(
    2\pi\lambda^\star
    +
    C_3^{(p_{rx})}
    +
    \mathcal D_1^{(p_{rx})}\left(\lambda^\star\right)
    \right)
    +
    \mathcal D_0^{(p_{rx})}\left(\lambda^\star\right)
    }.
    \]
    \vspace{-0.08cm}
    \[
    \widehat{p}_{vy}\left(\lambda^\star\right)=
    \frac{
    C_0^{(p_{vy})}
    }{
    C_1^{(p_{vy})}\lambda^\star
    +
    C_2^{(p_{vy})}
    +
    \sin\left(
    2\pi\lambda^\star
    +
    C_3^{(p_{vy})}
    +
    \mathcal D_1^{(p_{vy})}\left(\lambda^\star\right)
    \right)
    +
    \mathcal D_0^{(p_{vy})}\left(\lambda^\star\right)
    }.
    \]
    \[
    \widehat{p}_{ry}\left(\lambda^\star\right)=
    \frac{
    C_0^{(p_{ry})}
    +
    C_1^{(p_{ry})}
    \cos\left(
    2\pi\lambda^\star
    +
    \mathcal D_0^{(p_{ry})}\left(\lambda^\star\right)
    \right)
    }{
    \left(\lambda^\star\right)^2
    +
    C_2^{(p_{ry})}\lambda^\star
    +
    C_3^{(p_{ry})}
    }.
    \]
    \vspace{-0.08cm}
    \[
    \widehat{p}_{vx}\left(\lambda^\star\right)=
    \frac{
    C_0^{(p_{vx})}
    +
    C_1^{(p_{vx})}
    \cos\left(
    2\pi\lambda^\star
    +
    \mathcal D_0^{(p_{vx})}\left(\lambda^\star\right)
    \right)
    }{
    \left(\lambda^\star\right)^2
    +
    C_2^{(p_{vx})}\lambda^\star
    +
    C_3^{(p_{vx})}
    }.
    \]
    \begin{itemize}
        \item \(\mathcal D_m^{(X)}\) -- поправочные функции в формулах управления.
        \item Числовые коэффициенты и развёрнутые \(\mathcal D\)-функции приведены в приложениях диссертации.
    \end{itemize}
    \begin{newresult}
        Формулы начального вектора сопряжённых переменных.
    \end{newresult}
\end{frame}

\begin{frame}[t]
    \frametitle{Конечные формулы для начального вектора: зависимость от \(\kappa\)}
    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.38\textwidth}
            \centering
            \includegraphics[width=\linewidth,height=0.29\textheight,keepaspectratio]{main_p_isolines}

            \vspace{0.06cm}
            \includegraphics[width=\linewidth,height=0.29\textheight,keepaspectratio]{main_p_error_heatmap}
        \end{column}
        \begin{column}{0.58\textwidth}
            \tiny
            \[
            \mathcal P_i^{(\kappa)}
            =
            \mathcal P_i^{(\kappa)}(\lambda^\star,\kappa_1,\kappa_2).
            \]
            \[
            \widehat p_{rx}(\lambda^\star,\kappa)
            =
            \ln\kappa
            \left(
            \frac{\mathcal P_{0}^{(\kappa)}}{\kappa}
            +
            \frac{\mathcal P_{1}^{(\kappa)}}{\kappa^{1/4}}
            \right).
            \]
            \[
            \widehat p_{vy}(\lambda^\star,\kappa)
            =
            \ln\kappa
            \left(
            \frac{\mathcal P_{6}^{(\kappa)}}{\kappa}
            +
            \frac{\mathcal P_{7}^{(\kappa)}}{\kappa^{1/4}}
            \right).
            \]
            \[
            \widehat p_{ry}(\lambda^\star,\kappa)
            =
            \ln\kappa
            \left(
            \frac{\mathcal P_{2}^{(\kappa)}}{\kappa}
            +
            \frac{\mathcal P_{3}^{(\kappa)}}{\kappa^{3/2}}
            \right).
            \]
            \[
            \widehat p_{vx}(\lambda^\star,\kappa)
            =
            \ln\kappa
            \left(
            \frac{\mathcal P_{4}^{(\kappa)}}{\kappa}
            +
            \frac{\mathcal P_{5}^{(\kappa)}}{\kappa^{3/2}}
            \right).
            \]
            \[
            \widehat{\mathbf z}^{\mathrm{pl}}_0(\lambda^\star,\kappa)
            =
            \left(
            \widehat p_{rx},
            \widehat p_{ry},
            0,
            \widehat p_{vx},
            \widehat p_{vy},
            0
            \right)^T.
            \]
            \begin{itemize}
                \item Используются два характерных значения \(\kappa_1=0.72\), \(\kappa_2=1.52\).
                \item Функции \(\mathcal P_i^{(\kappa)}\) являются вспомогательными функциями согласования компонент по \(\kappa\).
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}
