\begin{frame}[t]
    \frametitle{Конечные формулы для синодической фазы}
        \presentationsectionmark{Глава 2}
    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.40\textwidth}
            \centering
            \includegraphics[height=0.4\textheight]{main_phase_isolines}
            \vspace{0.06cm}
            \includegraphics[height=0.4\textheight]{main_phase_error_heatmap}
        \end{column}
        \begin{column}{0.56\textwidth}
            \presentationcompactsize
            \[
            \Psi =
            \lambda^\star-\frac{T}{\kappa^{3/2}},
            \qquad
            \widehat{\Psi}_{T}
            =
            \lambda^\star-\frac{\widehat{T}}{\kappa^{3/2}}.
            \]
            \[
            \widehat{\Psi}_{T}\left(\lambda^\star,\kappa\right)=
            \left(
            1-\frac{1}{\sqrt{\kappa}}
            +\frac{\sqrt6}{9}\frac{\ln\kappa}{\sqrt{\kappa}}
            \right)\lambda^\star
            -
            \frac{2}{27}\frac{\ln\kappa}{\sqrt{\kappa}}.
            \]
            \[
            \widehat{\Psi}\left(\lambda^\star,\kappa\right)
            =
            \mathcal A_0^{(\Psi)}(\kappa)
            +
            \mathcal A_1^{(\Psi)}(\kappa)\lambda^\star .
            \]
            \[
            \begin{aligned}
            \mathcal A_0^{(\Psi)}(\kappa)
            &=
            -C_0^{(\Psi)} e^{-\kappa}\ln^2\kappa,\\
            \mathcal A_1^{(\Psi)}(\kappa)
            &=
            C_0^{(\Psi)}
            \left(e^\kappa+1\right)e^{-\kappa}\ln\kappa,
            \end{aligned}
            \]
            \presentationtextsize
            \begin{itemize}
                \item \(\Psi\) -- нормированная непрерывная синодическая фаза
                \item \(\widehat{\Psi}_T\) строится через \(\widehat T\) и наследует ошибку этой формулы
                \item Для выбора точки оптимальной дальности используется самостоятельная формула \(\widehat{\Psi}\)
                \item \(\widehat T\) и \(\widehat\Psi\) позволяют выбрать точку оптимальной дальности
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}[t]
    \frametitle{Конечные формулы для начальных значений сопряжённых переменных}
        \presentationsectionmark{Глава 2}
    \presentationtextsize
    \vspace{-0.08cm}
    \[
    \widehat{\mathbf z}_0=
    \left(
    \widehat{p}_{rx},
    \widehat{p}_{ry},
    0,
    \widehat{p}_{vx},
    \widehat{p}_{vy},
    0
    \right).
    \]
    \vspace{-0.08cm}
    \[
    \widehat{p}_{rx}\left(\lambda^\star\right)=
    \frac{
    C_0^{(p_{rx})}
    }{
    C_1^{(p_{rx})}\lambda^\star
    +
    C_2^{(p_{rx})}
    +
    \sin\left(
    2\pi\lambda^\star
    +
    C_3^{(p_{rx})}
    +
    \mathcal D_1^{(p_{rx})}\left(\lambda^\star\right)
    \right)
    +
    \mathcal D_0^{(p_{rx})}\left(\lambda^\star\right)
    }.
    \]
    \vspace{-0.08cm}
    \[
    \widehat{p}_{vy}\left(\lambda^\star\right)=
    \frac{
    C_0^{(p_{vy})}
    }{
    C_1^{(p_{vy})}\lambda^\star
    +
    C_2^{(p_{vy})}
    +
    \sin\left(
    2\pi\lambda^\star
    +
    C_3^{(p_{vy})}
    +
    \mathcal D_1^{(p_{vy})}\left(\lambda^\star\right)
    \right)
    +
    \mathcal D_0^{(p_{vy})}\left(\lambda^\star\right)
    }.
    \]
    \[
    \widehat{p}_{ry}\left(\lambda^\star\right)=
    \frac{
    C_0^{(p_{ry})}
    +
    C_1^{(p_{ry})}
    \cos\left(
    2\pi\lambda^\star
    +
    \mathcal D_0^{(p_{ry})}\left(\lambda^\star\right)
    \right)
    }{
    \left(\lambda^\star\right)^2
    +
    C_2^{(p_{ry})}\lambda^\star
    +
    C_3^{(p_{ry})}
    }.
    \]
    \vspace{-0.08cm}
    \[
    \widehat{p}_{vx}\left(\lambda^\star\right)=
    \frac{
    C_0^{(p_{vx})}
    +
    C_1^{(p_{vx})}
    \cos\left(
    2\pi\lambda^\star
    +
    \mathcal D_0^{(p_{vx})}\left(\lambda^\star\right)
    \right)
    }{
    \left(\lambda^\star\right)^2
    +
    C_2^{(p_{vx})}\lambda^\star
    +
    C_3^{(p_{vx})}
    }.
    \]
    \presentationtextsize
    \begin{itemize}
        \item \(\mathcal D_m^{(X)}\) -- поправочные функции в формулах управления
        \item Компоненты \((p_{rx},p_{vy})\) и \((p_{ry},p_{vx})\) образуют две естественные пары
    \end{itemize}
\end{frame}
