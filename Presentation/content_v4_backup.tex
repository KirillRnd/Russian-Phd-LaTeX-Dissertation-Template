\section{Модельное многообразие}

\begin{frame}
    \frametitle{ИРТ как модельная краевая задача и параметры}
    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.47\textwidth}
            \small
            \[
                J=\int_{t_0}^{t_f}\|\mathbf a(t)\|^2\,dt
            \]
            \[
                \mathbf y_0=
                \begin{pmatrix}
                \mathbf r_0,\mathbf v_0\\
                \mathbf p_{r0},\mathbf p_{v0}
                \end{pmatrix}
            \]
            \[
                (\lambda,\kappa)
                \longrightarrow
                T,\ \Psi,\ J,\ \mathbf z_0
            \]
            \begin{itemize}
                \item \(\kappa\) задаёт отношение радиусов круговых орбит;
                \item \(\lambda\) задаёт витковую дальность;
                \item синодическая фаза связывает ветвь с датой старта.
            \end{itemize}
        \end{column}
        \begin{column}{0.49\textwidth}
            \centering
            \includegraphics[height=0.68\textheight]{virtual_planet}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}
    \frametitle{Модельное многообразие и дискретное продолжение}
    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.61\textwidth}
            \centering
            \includegraphics[width=\linewidth]{continuation_paths}
        \end{column}
        \begin{column}{0.35\textwidth}
            \small
            \begin{itemize}
                \item базовое решение продолжается малыми шагами по пространству параметров;
                \item соседние решения дают устойчивый начальный вектор для следующей КЗ;
                \item границы регулярной области отделяются вычислительными барьерами.
            \end{itemize}

\begin{frame}
    \frametitle{От многообразия к конечным формулам}
    \begin{columns}[T,onlytextwidth]

            \scriptsize
            \[
            \begin{array}{c}
            \text{регулярные ветви решений}\\
            \Downarrow\\
            \text{выбор структуры}\\
            \text{аппроксимаций}\\
            \Downarrow\\
            \text{оценка коэффициентов}\\
            \Downarrow\\
            \widehat T,\ \widehat\Psi,\ \widehat J,\ \widehat{\mathbf z}_0
            \end{array}
            \]
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{itemize}
                \item формулы строятся для регулярных областей модельного многообразия;
                \item проверяются точность и отсутствие сингулярностей;
                \item результат используется не вместо численного решения, а как вход в уточнение краевой задачи.
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}

\section{Конечные формулы}

\begin{frame}
    \frametitle{КЛО и точки оптимальной дальности}
    \begin{columns}[T,onlytextwidth]

            \quad\cap\quad

            \]

                \item дата старта задаёт начальную синодическую фазу;
                \item изолиния фазы пересекает КЛО;
                \item точка пересечения выбирает ветвь и дальность.
            \end{itemize}

                \item \(\kappa\) задаётся целевой орбитой;

        \end{column}

\section{R-преобразования}

\begin{frame}
    \frametitle{Смысл \(R\)-преобразования}
    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.42\textwidth}
            \small
            \[
            \begin{array}{c}
            \mathrm{КЗ}_1\rightarrow\mathrm{экстремаль}_1\\
            \xrightarrow{R}\\
            \mathrm{экстремаль}_2\rightarrow\mathrm{КЗ}_2
            \end{array}
            \]
            \begin{itemize}
                \item преобразуется начальный вектор расширенной системы;
                \item меняются параметры краевой задачи;
                \item результат используется как начальный вектор для новой постановки.
            \end{itemize}
        \end{column}
        \begin{column}{0.54\textwidth}
            \centering
            \includegraphics[width=\linewidth]{R_small_change_demo}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}
    \frametitle{\(R\)-преобразование I типа}
    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.41\textwidth}
            \small
            \begin{itemize}
                \item задаёт переход к новому начальному фазовому вектору;
                \item согласованно преобразует начальные сопряжённые переменные;
                \item отличается от простого поворота вектора сопряжённых переменных.
            \end{itemize}
        \end{column}
        \begin{column}{0.55\textwidth}
            \centering
            \includegraphics[width=\linewidth]{R_type_I_demo}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}
    \frametitle{\(R\)-преобразование II типа}
    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.38\textwidth}
            \small
            \[
            (e,\omega)
            \longrightarrow
            (\zeta,\chi)
            \longrightarrow
            R_{e\omega}
            \]
            \begin{itemize}
                \item задаются эксцентриситет и аргумент перицентра;
                \item учитывается деформация витковой дальности;
                \item переход используется для эллиптической конечной орбиты.
            \end{itemize}
        \end{column}
        \begin{column}{0.58\textwidth}
            \centering
            \includegraphics[width=\linewidth]{R_type_II_eomega_example}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}
    \frametitle{\(R\)-преобразование III типа}
    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.38\textwidth}
            \small
            \[
            (i,\Omega)
            \longrightarrow
            (j,\gamma)
            \longrightarrow

                \item задаются наклонение и долгота восходящего узла;
                \item плоская модель переводится в пространственную постановку;
                \item сохраняется вычислительная роль начального вектора.

        \end{column}

\section{Алгоритм и проверка}

\begin{frame}
    \frametitle{Управляющие параметры и алгоритм}
    \small
    \[
    t_0,\ \text{целевая орбита}
    \rightarrow
    \Phi,\ (a,e,\omega,i,\Omega)
    \rightarrow
    \boldsymbol q
    \rightarrow
    \widehat{\mathbf z}_0
    \rightarrow
    R_\Sigma
    \rightarrow
    \mathbf y_0^{init}
    \rightarrow
    \text{КЗ}
    \]
    \begin{center}
        \footnotesize
        \begin{tabular}{@{}p{0.27\linewidth}p{0.62\linewidth}@{}}
            \toprule
            Параметры & Роль \\
            \midrule
            \(\lambda,\kappa\) & точка на модельном многообразии \\
            \(\zeta,\chi\) & форма конечной орбиты, \(R\)-II \\
            \(j,\gamma\) & пространственная ориентация, \(R\)-III \\
            \bottomrule
        \end{tabular}
    \end{center}
    Вектор \(\boldsymbol q\) собирает модельные формулы и параметры прикладной орбиты в единый вычислительный интерфейс.
\end{frame}
