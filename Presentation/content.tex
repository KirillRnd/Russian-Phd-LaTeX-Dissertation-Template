\begin{frame}
    \frametitle{Цель и задачи}
    \textbf{Цель:} разработать и апробировать численно-аналитический метод построения начальных приближений для энергетически оптимальных межпланетных перелётов с непрерывной тягой.

    \medskip
    \begin{enumerate}
        \item построить и исследовать модельное многообразие решений;

        \item разработать линейные \(R\)-преобразования сопряжённых переменных;
        \item построить алгоритм начального приближения и проверить его на прикладных перелётах.
    \end{enumerate}
\end{frame}

\section{R-преобразования}

\begin{frame}
    \frametitle{Переход к другой геометрии и смысл \(R\)}
    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.48\textwidth}
            \[
            \mathrm{КЗ}_1 \rightarrow
            \mathrm{экстремаль}_1 \xrightarrow{R}
            \mathrm{экстремаль}_2 \rightarrow
            \mathrm{КЗ}_2
            \]
            \begin{itemize}
                \item Меняется сама краевая задача.
                \item Преобразуется начальный вектор расширенной системы.
                \item Оптимальность не переносится в старую задачу, а задаётся новая экстремаль.
            \end{itemize}
        \end{column}
        \begin{column}{0.48\textwidth}
            \centering
            \includegraphics[width=\linewidth]{R_small_change_demo}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}
    \frametitle{\(R\)-преобразование I типа}
    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.48\textwidth}
            \begin{itemize}
                \item Согласованно меняет начальный фазовый вектор и начальный вектор сопряжённых переменных.
                \item Используется для перехода к заданному начальному состоянию из эфемерид.
                \item Отличается от простого поворота вектора сопряжённых переменных.
            \end{itemize}
        \end{column}

\begin{frame}
    \frametitle{\(R\)-преобразование II типа}
    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.44\textwidth}
            \[
            (e,\omega)\longrightarrow(\zeta,\chi)
            \longrightarrow R_{e\omega}
            \]
            \begin{itemize}
                \item \(e\) задаётся управляющим параметром \(\zeta\);
                \item \(\omega\) задаётся углом \(\chi\);
                \item побочная деформация витковой дальности учитывается далее в алгоритме.
            \end{itemize}
        \end{column}

    \frametitle{\(R\)-преобразование III типа}
    \begin{columns}[T,onlytextwidth]

            (i,\Omega)\longrightarrow(j,\gamma)

\section{Алгоритм и апробация}

\begin{frame}
    \frametitle{Управляющие параметры \(\boldsymbol q\)}
    \[
    \boldsymbol q=(\lambda,\kappa,\zeta,\chi,j,\gamma)
    \]
    \begin{center}
        \small
        \begin{tabular}{cll}
            \toprule
            Параметр & Роль & Источник \\
            \midrule
            \(\lambda,\kappa\) & точка на модельном многообразии & глава 2 \\
            \(\zeta,\chi\) & эксцентриситет и аргумент перицентра & \(R\)-II \\
            \(j,\gamma\) & наклонение и долгота узла & \(R\)-III \\
            \bottomrule
        \end{tabular}
    \end{center}
    \begin{itemize}
        \item Это вычислительный интерфейс, а не новый набор орбитальных элементов.
        \item Через \(\boldsymbol q\) конечные формулы и \(R\)-преобразования соединяются в одну процедуру.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Алгоритм построения начального приближения}
    \small
    \[
    t_0,\ \mathrm{target\ orbit}
    \rightarrow
    \Phi,\ (a,e,\omega,i,\Omega)
    \rightarrow
    \boldsymbol q
    \rightarrow
    \widehat{\mathbf z}_0^{\mathrm{pl}}
    \xrightarrow{R_{\Sigma}}
    \mathbf y_0^{\mathrm{init}}
    \rightarrow
    \mathrm{BVP}
    \]
    \begin{enumerate}
        \item Из эфемерид берутся начальный фазовый вектор и целевая орбита.
        \item По фазе выбирается допустимая точка оптимальной дальности.
        \item Конечные формулы дают \(\widehat T\), \(\widehat J\), \(\widehat{\mathbf z}_0^{\mathrm{pl}}\).
        \item Композиция \(R\)-преобразований переносит приближение в прикладную постановку.
    \end{enumerate}
\end{frame}
