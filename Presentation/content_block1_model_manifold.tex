\section{Введение}

\begin{frame}
    \frametitle{Актуальность}
    \presentationtextsize
    \begin{itemize}
        \item Межпланетные перелёты с электрореактивной тягой формируются в результате длительного управляемого движения.
        \item На ранних этапах баллистического проектирования требуется многократно оценивать даты старта, продолжительность перелёта, расход рабочего тела и параметры управления.
        \item Для численного уточнения краевой задачи требуется начальное приближение, согласованное с выбранной ветвью решения.
    \end{itemize}

    \vspace{0.2cm}
    \begin{block}{Проектная цепочка}
        \presentationcompactsize
        Дата старта и целевая орбита определяют выбор ветви решения; затем оцениваются продолжительность перелёта, значение энергетического критерия и начальное приближение, после чего выполняется численное уточнение краевой задачи.
    \end{block}
\end{frame}

\begin{frame}
    \frametitle{Цель и задачи}
    \presentationtextsize
    \textbf{Цель:} разработать и апробировать численно-аналитический метод построения начальных приближений для рассматриваемого класса межпланетных перелётов с непрерывной тягой. В модельной постановке оптимальность задаётся квадратичным функционалом управления. Итогом метода является алгоритм выбора такого приближения; модельное многообразие, конечные формулы и \(R\)-преобразования служат его конструктивной основой.

    \medskip
    Для достижения цели решаются задачи:
    \begin{enumerate}
        \item построить регулярное модельное многообразие как основу последующей параметризации;
        \item получить конечные формулы для характеристик экстремалей и начальных значений сопряжённых переменных;
        \item разработать линейные преобразования начальных значений сопряжённых переменных для перехода к эллиптической и пространственной конечной орбите;
        \item собрать эти результаты в алгоритм выбора начального приближения и проверить его на прикладных классах перелётов.
    \end{enumerate}
\end{frame}

\section{Модельное многообразие}

\begin{frame}[t]
    \frametitle{Идеально регулируемая тяга (ИРТ) как модельная краевая задача}
    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.62\textwidth}
            \presentationcompactsize
            \[
            J=\frac{1}{2}\int\limits_{t_0}^{t_f}\|\mathbf a(t)\|^2\,dt \to \min,
            \qquad
            H=\mathbf p_r^{T}\mathbf v+
            \mathbf p_v^{T}\left(-\frac{\mathbf r}{r^3}+\mathbf a\right)
            -\frac{1}{2}\mathbf a^{T}\mathbf a.
            \]
            \[
            \frac{\partial H}{\partial \mathbf a}=0
            \quad\Rightarrow\quad
            \mathbf a=\mathbf p_v.
            \]
            \[
            \left\{
            \begin{aligned}
            \dot{\mathbf r} &= \mathbf v,\\
            \dot{\mathbf v} &= -\frac{\mathbf r}{r^3}+\mathbf p_v,\\
            \dot{\mathbf p}_r
            &= \frac{1}{r^3}\mathbf p_v
            -\frac{3}{r^5}(\mathbf r\cdot\mathbf p_v)\mathbf r,\\
            \dot{\mathbf p}_v &= -\mathbf p_r.
            \end{aligned}
            \right.
            \]
        \end{column}
        \begin{column}{0.34\textwidth}
            \centering
            \includegraphics[width=0.52\linewidth,height=0.28\textheight,keepaspectratio]{virtual_planet}

            \vspace{-0.03cm}
            \presentationcompactsize
            \[
            \begin{aligned}
            \lambda&=\frac{\varphi}{2\pi},\\
            \mathbf r_0&=(1,0,0)^{T},\\
            \mathbf v_0&=(0,1,0)^{T},\\
            \mathbf r_f&=\kappa(\cos 2\pi\lambda,\sin 2\pi\lambda,0)^{T},\\
            \mathbf v_f&=\kappa^{-1/2}(-\sin 2\pi\lambda,\cos 2\pi\lambda,0)^{T}.
            \end{aligned}
            \]
            \[
            \mathbf z_0=\left(\mathbf p_{r0},\mathbf p_{v0}\right),
            \]
            \begin{itemize}
                \item \(\lambda\) задаёт витковую дальность.
                \item \(\mathbf r_f,\mathbf v_f\) задаются через \(\lambda,\kappa\).
                \item \(J\) -- квадратичный функционал; \(\mathbf z_0\) -- начальные значения \(\mathbf p_r,\mathbf p_v\).
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}[t]
    \frametitle{Модельное многообразие и дискретное продолжение}
    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.58\textwidth}
            \centering
            \includegraphics[height=0.48\textheight]{continuation_paths}
        \end{column}
        \begin{column}{0.38\textwidth}
            \presentationcompactsize
            \[
            \boldsymbol\xi=(\lambda,\kappa).
            \]
            \[
            \begin{gathered}
            M(\lambda,\kappa):(\lambda,\kappa)\mapsto\\
            \bigl(\mathbf z_0,T,\Psi,J\bigr)(\lambda,\kappa).
            \end{gathered}
            \]
            \begin{itemize}
                \item Модельное многообразие понимается как образ отображения, построенного на одной регулярной ветви решений.
                \item \(\lambda,\kappa\) являются входными параметрами плоской круговой модели.
                \item \(T,\Psi,\mathbf z_0\) являются характеристиками построенной экстремали; \(J\) -- значение квадратичного функционала.
                \item Базовое решение используется как исходная точка для продолжения по сетке параметров.
                \item Соседние решения дают начальное приближение для следующей краевой задачи.
                \item Вычислительные барьеры отделяют границы устойчивого локального продолжения.
            \end{itemize}
        \end{column}
    \end{columns}
    \begin{newresult}
        Построение регулярного модельного многообразия.
    \end{newresult}
\end{frame}

\begin{frame}
    \frametitle{От многообразия к конечным формулам}
    \presentationtextsize
    \[
    M(\lambda,\kappa)
    \longrightarrow
    \widehat T,\ \widehat\Psi,\ \widehat J,\ \widehat{\mathbf z}_0.
    \]
    \begin{itemize}
        \item Механическая структура задаётся ИРТ-КЗ, расширенной системой и параметрами \((\lambda,\kappa)\).
        \item Численные решения анализируются в параметрических представлениях, согласованных с этой структурой.
        \item Для регулярных зависимостей строятся непосредственные аппроксимации; для более сложных -- иерархические коэффициент-функции.
        \item Символьная регрессия применяется как инструмент компактной записи интерпретируемых формул.
        \item Качество формул проверяется метриками ошибок и краевыми невязками после интегрирования расширенной системы.
    \end{itemize}
    \begin{newresult}
        Конечные формулы экстремалей, полученные после построения модельной структуры.
    \end{newresult}
\end{frame}

\begin{frame}[t]
    \frametitle{КЛО и точки оптимальной дальности}
    \begin{columns}[T,onlytextwidth]

            \includegraphics[height=0.48\textheight]{Isolines}

            \presentationtextsize
            \[
            \Phi=\mathrm{const}
            \quad\cap\quad
            \mathrm{КЛО}
            \quad\Rightarrow\quad
            \lambda^\star.
            \]
            \[
            \varphi=\Phi+\frac{T}{\kappa^{3/2}}.
            \]
            \begin{itemize}
                \item Дата старта задаёт начальную синодическую фазу.
                \item КЛО связывает структуру ветвей с выбором \(\lambda^\star\).
                \item Изолиния фиксированной фазы пересекает КЛО в точке оптимальной дальности.
                \item Точка оптимальной дальности задаёт витковую дальность выбранной ветви.
            \end{itemize}
        \end{column}
    \end{columns}
    \begin{newresult}
        КЛО и точки оптимальной дальности переводят начальную фазу в выбор крупномасштабной ветви.
    \end{newresult}
\end{frame}
