\begin{frame}[t]
    \frametitle{Управляющие параметры как вычислительный интерфейс}
    \presentationtextsize
    \[
    \boldsymbol q =
    \left(
    \lambda,\kappa,\zeta,\chi,j,\gamma
    \right).
    \]
    \[
    (\lambda,\kappa)
    \longrightarrow
    \widehat T,\quad
    \widehat J,\quad
    \widehat{\mathbf z}^{\mathrm{pl}}_0,
    \qquad
    (e,\omega)
    \longrightarrow
    (\zeta,\chi)
    \longrightarrow
    R_{e\omega},
    \]
    \[
    (i,\Omega)
    \longrightarrow
    (j,\gamma)
    \longrightarrow
    R_{i\Omega},
    \qquad
    R_{\Sigma}
    =
    R_{i\Omega}
    R_{e\omega}
    R_0 .
    \]
    \begin{center}
        \begin{tabular}{@{}>{\raggedright\arraybackslash}p{0.23\linewidth}>{\raggedright\arraybackslash}p{0.36\linewidth}>{\raggedright\arraybackslash}p{0.27\linewidth}@{}}
            \toprule
            Параметры & Роль & Источник \\
            \midrule
            \(\lambda,\kappa\) & точка на модельном многообразии & формулы главы 2 \\
            \(\zeta,\chi\) & форма конечной орбиты: \(e,\omega\) & \(R\)-II \\
            \(j,\gamma\) & пространственная ориентация: \(i,\Omega\) & \(R\)-III \\
            \bottomrule
        \end{tabular}
    \end{center}
    \begin{itemize}
        \item \(\Omega\) -- орбитальный элемент конечной орбиты.
        \item \(\gamma\) -- управляющий угловой параметр матрицы преобразования.
    \end{itemize}
    \begin{newresult}
        Единый интерфейс управляющих параметров.
    \end{newresult}
\end{frame}

\begin{frame}[t]
    \frametitle{Алгоритм построения начального приближения}
    \presentationtextsize
    \[
    t_0,\ \text{целевая орбита}
    \longrightarrow
    \Phi,\ (a_{\mathrm{orb}},e,\omega,i,\Omega)
    \longrightarrow
    \boldsymbol q.
    \]
    \[
    \begin{gathered}
    \lambda_{\mathrm{corr}}=\lambda^\star-\Delta\lambda,\\
    \text{коррекция витковой дальности при }R\text{-II}
    \end{gathered}
    \qquad
    (\lambda_{\mathrm{corr}},\kappa)
    \longrightarrow
    \widehat T,\quad
    \widehat J,\quad
    \widehat{\mathbf z}^{\mathrm{pl}}_0 .
    \]
    \[
    \widehat{\mathbf z}^{\mathrm{pl}}_0=
    \left(
    \widehat{\mathbf p}_{r0}^{\mathrm{pl}},
    \widehat{\mathbf p}_{v0}^{\mathrm{pl}}
    \right).
    \]
    \[
    \mathbf p_{r0}^{\mathrm{init}}
    =
    (R_{\Sigma}^{T})^{-1}
    \widehat{\mathbf p}_{r0}^{\mathrm{pl}},
    \qquad
    \mathbf p_{v0}^{\mathrm{init}}
    =
    (R_{\Sigma}^{T})^{-1}
    \widehat{\mathbf p}_{v0}^{\mathrm{pl}}.
    \]
    \[
    \mathbf y_0^{\mathrm{init}}
    =
    \left(
    \mathbf r_0,
    \mathbf v_0,
    \mathbf p_{r0}^{\mathrm{init}},
    \mathbf p_{v0}^{\mathrm{init}}
    \right).
    \]
    \begin{enumerate}
        \item По дате старта определяется начальная синодическая фаза.
        \item По КЛО и изолинии фазы выбирается \(\lambda^\star\).
        \item По формулам главы 2 вычисляются \(\widehat T\), \(\widehat J\), \(\widehat{\mathbf z}^{\mathrm{pl}}_0\).
        \item По формулам главы 3 строится \(R_\Sigma\).
        \item Получается начальный вектор расширенной системы \(\mathbf y_0^{\mathrm{init}}\).
    \end{enumerate}
    \begin{newresult}
        Алгоритм построения начального вектора.
    \end{newresult}
\end{frame}
