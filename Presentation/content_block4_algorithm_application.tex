\section{Алгоритм и проверка}

\begin{frame}
    \frametitle{Управляющие параметры начального приближения}
    \scriptsize
    \[
    \boldsymbol q =
    \left(
    \lambda,\kappa,\zeta,\chi,j,\gamma
    \right).
    \]
    \[
    (\lambda,\kappa)
    \longrightarrow
    \widehat T,\quad
    \widehat J,\quad
    \widehat{\mathbf z}^{\mathrm{pl}}_0,
    \qquad
    (e,\omega)
    \longrightarrow
    (\zeta,\chi)
    \longrightarrow
    R_{e\omega},
    \]
    \[
    (i,\Omega)
    \longrightarrow
    (j,\gamma)
    \longrightarrow
    R_{i\Omega},
    \qquad
    R_{\Sigma}
    =
    R_{i\Omega}
    R_{e\omega}
    R_0 .
    \]
    \begin{center}
        \begin{tabular}{@{}p{0.26\linewidth}p{0.38\linewidth}p{0.24\linewidth}@{}}
            \toprule
            Параметры & Роль & Источник \\
            \midrule
            \(\lambda,\kappa\) & точка на модельном многообразии & конечные формулы главы 2 \\
            \(\zeta,\chi\) & форма конечной орбиты: \(e,\omega\) & \(R\)-II \\
            \(j,\gamma\) & пространственная ориентация: \(i,\Omega\) & \(R\)-III \\
            \bottomrule
        \end{tabular}
    \end{center}
    \begin{itemize}
        \item \(\Omega\) -- орбитальный элемент конечной орбиты.
        \item \(\gamma\) -- управляющий угловой параметр матрицы преобразования.
    \end{itemize}
    \begin{block}{Новый результат}
        \scriptsize Введён вектор управляющих параметров \(\boldsymbol q\) как вычислительный интерфейс для построения начального приближения.
    \end{block}
\end{frame}

\begin{frame}
    \frametitle{Алгоритм построения начального приближения}
    \scriptsize
    \[
    t_0,\ \text{целевая орбита}
    \longrightarrow
    \Phi,\ (a,e,\omega,i,\Omega)
    \longrightarrow
    \boldsymbol q.
    \]
    \[
    \lambda_{\mathrm{corr}}=\lambda^\star-\Delta\lambda,
    \qquad
    (\lambda_{\mathrm{corr}},\kappa)
    \longrightarrow
    \widehat T,\quad
    \widehat J,\quad
    \widehat{\mathbf z}^{\mathrm{pl}}_0 .
    \]
    \[
    \widehat{\mathbf z}^{\mathrm{pl}}_0=
    \left(
    \widehat{\mathbf p}_{r0}^{\mathrm{pl}},
    \widehat{\mathbf p}_{v0}^{\mathrm{pl}}
    \right).
    \]
    \[
    \mathbf p_{r0}^{\mathrm{init}}
    =
    (R_{\Sigma}^{T})^{-1}
    \widehat{\mathbf p}_{r0}^{\mathrm{pl}},
    \qquad
    \mathbf p_{v0}^{\mathrm{init}}
    =
    (R_{\Sigma}^{T})^{-1}
    \widehat{\mathbf p}_{v0}^{\mathrm{pl}}.
    \]
    \[
    \mathbf y_0^{\mathrm{init}}
    =
    \left(
    \mathbf r_0,
    \mathbf v_0,
    \mathbf p_{r0}^{\mathrm{init}},
    \mathbf p_{v0}^{\mathrm{init}}
    \right).
    \]
    \begin{enumerate}
        \item По дате старта определяется начальная синодическая фаза.
        \item По КЛО и изолинии фазы выбирается \(\lambda^\star\).
        \item По формулам главы 2 вычисляются \(\widehat T\), \(\widehat J\), \(\widehat{\mathbf z}^{\mathrm{pl}}_0\).
        \item По формулам главы 3 строится \(R_\Sigma\).
        \item Получается начальный вектор расширенной системы \(\mathbf y_0^{\mathrm{init}}\).
    \end{enumerate}
    \begin{block}{Новый результат}
        \scriptsize Разработан алгоритм построения начального вектора расширенной системы на основе конечных формул и композиции \(R\)-преобразований.
    \end{block}
\end{frame}

\begin{frame}[t]
    \frametitle{Сценарии использования и программная реализация}
    \tiny
    \[
    \begin{array}{c}
    \text{дата старта, целевая орбита}
    \rightarrow
    \boldsymbol q
    \rightarrow
    \text{ИРТ-приближение}\\
    \rightarrow
    \mathbf v_\infty
    \rightarrow
    \text{ПСИ-модель}
    \rightarrow
    \text{миссионный анализ}.
    \end{array}
    \]
    \vspace{-0.12cm}
    \begin{center}
        \begin{tabular}{@{}p{0.27\linewidth}p{0.36\linewidth}p{0.27\linewidth}@{}}
            \toprule
            Блок & Роль & Выход \\
            \midrule
            Начальное приближение & использует конечные формулы и \(R_\Sigma\) & \(\mathbf y_0^{\mathrm{init}},\ \widehat T,\ \widehat J\) \\
            Уточнение траектории & использует \(\boldsymbol q\) как переменные оптимизации & решение с малыми невязками правого конца \\
            Перенос между датами старта & применяет \(R\)-I к опорному решению & начальное приближение для новой даты \\
            Ненулевая отлётная скорость & продолжение от \(\mathbf v_\infty=0\) & решение с гиперболическим избытком \\
            Переход к ПСИ-модели & использует ИРТ-решение как гладкое приближение & опорная bang-bang траектория \\
            Массовый расчёт & повторяет перенос и уточнение на сетке дат & окна старта, длительность, \(\Delta V\), расход топлива \\
            \bottomrule
        \end{tabular}
    \end{center}
    \vspace{-0.12cm}
    \begin{itemize}
        \item Научный результат: конечные формулы, \(R\)-преобразования, управляющие параметры и алгоритм.
        \item Программная реализация: автоматизированная сборка этих элементов в воспроизводимую вычислительную цепочку.
    \end{itemize}
