\section{Алгоритм и прикладная проверка}

\begin{frame}[t]
    \frametitle{Управляющие параметры как вычислительный интерфейс}
    \presentationtextsize
    \[
    \boldsymbol q =
    \left(
    \lambda,\kappa,\zeta,\chi,j,\gamma
    \right).
    \]
    \[
    (\lambda,\kappa)
    \longrightarrow
    \widehat T,\quad
    \widehat J,\quad
    \widehat{\mathbf z}^{\mathrm{pl}}_0,
    \qquad
    (e,\omega)
    \longrightarrow
    (\zeta,\chi)
    \longrightarrow
    R_{e\omega},
    \]
    \[
    (i,\Omega)
    \longrightarrow
    (j,\gamma)
    \longrightarrow
    R_{i\Omega},
    \qquad
    R_{\Sigma}
    =
    R_{i\Omega}
    R_{e\omega}
    R_0 .
    \]
    \begin{center}
        \begin{tabular}{@{}>{\raggedright\arraybackslash}p{0.23\linewidth}>{\raggedright\arraybackslash}p{0.36\linewidth}>{\raggedright\arraybackslash}p{0.27\linewidth}@{}}
            \toprule
            Параметры & Роль & Источник \\
            \midrule
            \(\lambda,\kappa\) & выбор ветви и точки оптимальной дальности & формулы главы 2 \\
            \(\zeta,\chi\) & задание формы конечной орбиты: \(e,\omega\) & \(R\)-II \\
            \(j,\gamma\) & задание пространственной ориентации: \(i,\Omega\) & \(R\)-III \\
            \bottomrule
        \end{tabular}
    \end{center}
    \begin{itemize}
        \item \(\boldsymbol q\) параметризует переход от модельной краевой задачи к целевой краевой задаче.
        \item Уточнение в \(\boldsymbol q\) сохраняет связь с выбранной ветвью, конечными формулами и \(R\)-преобразованиями.
    \end{itemize}
    \begin{newresult}
        Вектор \(\boldsymbol q\) параметризует переход от модельной краевой задачи к целевой краевой задаче.
    \end{newresult}
\end{frame}

\begin{frame}[t]
    \frametitle{Алгоритм построения начального приближения}
    \presentationtextsize
    \[
    t_0,\ \text{целевая орбита}
    \longrightarrow
    \Phi_0
    \longrightarrow
    \lambda^\star,\ (a_{\mathrm{orb}},e,\omega,i,\Omega)
    \longrightarrow
    \boldsymbol q_{\mathrm{init}}.
    \]
    \[
    \begin{gathered}
    \lambda_{\mathrm{corr}}=\lambda^\star-\Delta\lambda,\\
    \text{коррекция витковой дальности при }R\text{-II}
    \end{gathered}
    \qquad
    (\lambda_{\mathrm{corr}},\kappa)
    \longrightarrow
    \widehat T,\quad
    \widehat J,\quad
    \widehat{\mathbf z}^{\mathrm{pl}}_0 .
    \]
    \[
    \widehat{\mathbf z}^{\mathrm{pl}}_0=
    \left(
    \widehat{\mathbf p}_{r0}^{\mathrm{pl}},
    \widehat{\mathbf p}_{v0}^{\mathrm{pl}}
    \right).
    \]
    \[
    \mathbf p_{r0}^{\mathrm{init}}
    =
    (R_{\Sigma}^{T})^{-1}
    \widehat{\mathbf p}_{r0}^{\mathrm{pl}},
    \qquad
    \mathbf p_{v0}^{\mathrm{init}}
    =
    (R_{\Sigma}^{T})^{-1}
    \widehat{\mathbf p}_{v0}^{\mathrm{pl}}.
    \]
    \[
    \mathbf y_0^{\mathrm{init}}
    =
    \left(
    \mathbf r_0,
    \mathbf v_0,
    \mathbf p_{r0}^{\mathrm{init}},
    \mathbf p_{v0}^{\mathrm{init}}
    \right).
    \]
    \[
    \left(\mathbf y_0^{\mathrm{init}},\widehat T\right)
    \longrightarrow
    \text{уточнение КЗ}.
    \]
    \begin{enumerate}
        \item По дате старта определяется начальная синодическая фаза.
        \item По КЛО и изолинии фазы выбирается \(\lambda^\star\).
        \item По формулам главы 2 вычисляются \(\widehat T\), \(\widehat J\), \(\widehat{\mathbf z}^{\mathrm{pl}}_0\).
        \item По формулам главы 3 строится \(R_\Sigma\).
        \item Получается начальное приближение для последующего уточнения.
    \end{enumerate}
    \begin{newresult}
        Разработан алгоритм построения начального приближения по дате старта и параметрам целевой орбиты.
    \end{newresult}
\end{frame}

\begin{frame}
    \frametitle{Уточнение в управляющих параметрах}
    \presentationtextsize
    \[
    \mathbf q=(\lambda,\kappa,\zeta,\chi,j,\gamma),
    \qquad
    \mathbf F(\mathbf q)=0.
    \]
    \[
    \mathbf F(\boldsymbol q)
    =
    \begin{pmatrix}
    \Delta e\\
    \Delta i\\
    \Delta\Omega\\
    \Delta\omega\\
    \Delta\varphi\\
    \Delta a_{\mathrm{orb}}
    \end{pmatrix}
    =0.
    \]
    \centering
    \textbf{Невязки после уточнения в параметрах \(\boldsymbol q\)}
    \begin{tabular}{@{}p{0.38\linewidth}cc@{}}
        \toprule
        Целевой параметр конечной орбиты & Земля--Марс & Земля--Венера \\
        \midrule
        Эксцентриситет, \(e\) & \(3.9\times10^{-15}\) & \(1.9\times10^{-12}\) \\
        Наклонение, \(i\) & \(2.1\times10^{-14}\) град. & \(2.9\times10^{-13}\) град. \\
        ДВУ, \(\Omega\) & \(5.7\times10^{-14}\) град. & \(4.7\times10^{-13}\) град. \\
        Аргумент перицентра, \(\omega\) & \(3.0\times10^{-13}\) град. & \(1.2\times10^{-13}\) град. \\
        Угловая дальность, \(\varphi\) & \(1.7\times10^{-13}\) град. & \(3.3\times10^{-16}\) град. \\

        \bottomrule
    \end{tabular}
    \begin{itemize}
        \item Уточнение выполняется в пространстве управляющих параметров \(\boldsymbol q\), сохраняя связь с выбранной модельной ветвью.
        \item \(\mathbf F\) -- вектор невязок целевых параметров конечной орбиты после интегрирования расширенной системы.
        \item Малые невязки подтверждают работоспособность параметризации; это не утверждение о глобальной оптимальности.
    \end{itemize}
\end{frame}

            \resizebox{\linewidth}{!}{$
            \begin{array}{c}
            \text{ИРТ-приближение}
            \longrightarrow
            \mathbf v_\infty
            \longrightarrow
            \text{ПСИ-модель}\\
            \longrightarrow
            \text{опорное решение}
            \longrightarrow
            \text{массовый расчёт по } t_0 .
            \end{array}
            $}
            \]
            \vspace{-0.10cm}
            \begin{itemize}
                \item Алгоритм не отменяет окна старта, но строит начальное приближение для заданной даты.
                \item Это важно для малых КА, запускаемых попутно, когда дата выведения задаётся основной полезной нагрузкой.
                \item Интервал: 01.01.2028--01.01.2037, шаг один день.
                \item На выходе: \(\Delta V\), гиперболический избыток и продолжительность перелёта как функции \(t_0\).
            \end{itemize}
        \end{column}
    \end{columns}

\begin{frame}[t]
    \frametitle{Сценарии использования и вычислительная цепочка}
    \presentationcompactsize
    \[
    \begin{array}{c}
    \text{модельное многообразие}
    \rightarrow
    \text{конечные формулы}
    \rightarrow
    R\text{-преобразования}\\
    \rightarrow
    \boldsymbol q
    \rightarrow
    \text{алгоритм}
    \rightarrow
    \text{ILTT Toolbox}.
    \end{array}
    \]
    \vspace{-0.12cm}
    \begin{center}
        \begin{tabular}{@{}>{\raggedright\arraybackslash}p{0.25\linewidth}>{\raggedright\arraybackslash}p{0.34\linewidth}>{\raggedright\arraybackslash}p{0.29\linewidth}@{}}
            \toprule
            Блок & Роль & Выход \\
            \midrule
            Начальное приближение & использует конечные формулы и \(R_\Sigma\) & \(\mathbf y_0^{\mathrm{init}},\ \widehat T,\ \widehat J\) \\
            Уточнение КЗ & выполняется в параметрах \(\boldsymbol q\) & малые невязки конечной орбиты \\
            Даты старта & строит приближение для заданного \(t_0\) & расчётная сетка окон старта \\
            \(v_\infty\) и ПСИ & численно продолжает ИРТ-решение & прикладная bang-bang траектория \\
            Массовый расчёт & повторяет перенос и уточнение & \(T\), \(\Delta V\), расход рабочего тела \\
            \bottomrule
        \end{tabular}
    \end{center}
    \vspace{-0.12cm}
    \begin{itemize}
        \item Научный результат: алгоритм для задач, где модельное многообразие описывает структуру решений.
        \item ILTT Toolbox -- зарегистрированная реализация этой цепочки, а не только иллюстрационный пример.
        \item \(\widehat J\) остаётся аппроксимацией квадратичного функционала; расход рабочего тела появляется в массовом расчёте после прикладного уточнения.
    \end{itemize}
\end{frame}
