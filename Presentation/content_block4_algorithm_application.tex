\begin{frame}[t]
    \frametitle{Алгоритм построения начального приближения}
    \presentationtextsize
    \begin{enumerate}
        \item По дате старта \(t_0\) определяется начальная синодическая фаза \(\Phi_0\)
        \item Из большой полуоси орбиты целевого небесного тела определяется \(\kappa\) 
        \item Из \(\Phi_0\) и \(\kappa\) определяется \(\lambda^\star\)
        \item По формулам главы 2 из \(\lambda^\star\) и \(\kappa\) вычисляются \(\widehat T\), \(\widehat{\mathbf z}_0\)
        \item По формулам главы 3 строится \(R_\Sigma\) как функция от \(i,\Omega,e, \omega,\mathbf r_0,\mathbf v_0\)
        \item Линейное преобразование \(R_\Sigma\) применяется к вектору \(\widehat{\mathbf z}_0\)
        \item Получается начальное приближение \(\mathbf y_0^{\mathrm{init}}\) для последующего уточнения
    \end{enumerate}
    \begin{newresult}
        Разработан алгоритм построения начального приближения по дате старта и параметрам целевой орбиты
    \end{newresult}
\end{frame}

    \mathbf q=(\lambda,\kappa,\zeta,\chi,j,\gamma),
    \qquad
    \mathbf F(\mathbf q)=0.

    \mathbf F(\boldsymbol q)

    \begin{pmatrix}
    \Delta e\\

    \Delta\varphi\\
